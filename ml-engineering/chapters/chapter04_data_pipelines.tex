\section{Data Pipeline Architecture}

\subsection{ETL vs ELT}

\textbf{ETL (Extract, Transform, Load):} Traditional approach where data is transformed before loading.

\textbf{ELT (Extract, Load, Transform):} Modern approach leveraging powerful data warehouses.

    'email_on_failure': True,
    'email_on_retry': False,
    'retries': 3,
    'retry_delay': timedelta(minutes=5),
}

def extract_data(**context):
    """Extract data from source"""
    # Implementation here
    pass

def transform_data(**context):
    """Transform and clean data"""
    # Implementation here
    pass

def validate_data(**context):
    """Validate data quality"""
    # Implementation here
    pass

with DAG(
    'ml_data_pipeline',
    default_args=default_args,
    description='ML data pipeline',
    schedule_interval='@daily',
    catchup=False
) as dag:

    extract = PythonOperator(
        task_id='extract_data',
        python_callable=extract_data
    )

    transform = PythonOperator(
        task_id='transform_data',
        python_callable=transform_data
    )

    validate = PythonOperator(
        task_id='validate_data',
        python_callable=validate_data
    )

    load = PostgresOperator(
        task_id='load_to_warehouse',
        postgres_conn_id='warehouse',
        sql='sql/load_data.sql'
    )

    extract >> transform >> validate >> load

\subsection{Real-Time Pipelines}

from kafka import KafkaConsumer
import json

consumer = KafkaConsumer(
    'ml_features',
    bootstrap_servers=['localhost:9092'],
    value_deserializer=lambda m: json.loads(m.decode('utf-8')),
    group_id='ml_pipeline'
)

for message in consumer:
    data = message.value
    # Process streaming data
    features = transform_features(data)
    # Store or use for inference
    save_to_feature_store(features)

\section{Data Pipeline Monitoring}

\subsection{Data Quality Checks}

\begin{lstlisting}[style=python]
import great_expectations as ge

# Load data as GE DataFrame
df = ge.read_csv('data.csv')

# Define expectations
df.expect_column_values_to_not_be_null('user_id')
df.expect_column_values_to_be_between('age', 0, 120)
df.expect_column_values_to_be_in_set('country', ['US', 'UK', 'CA'])

# Validate
results = df.validate()
print(results)
\end{lstlisting}
