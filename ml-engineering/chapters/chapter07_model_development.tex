\begin{lstlisting}[style=python, caption=Abstract base class for ML models]
# src/models/base.py
"""Base interfaces for ML models."""

from abc import ABC, abstractmethod
from typing import Any, Dict, Optional, Union
import numpy as np
import pandas as pd
from pathlib import Path
import joblib
import json


class BaseModel(ABC):
    """
    Abstract base class for all ML models.
    Enforces consistent interface across different model types.
    """

    def __init__(self, config: Dict[str, Any]):
        """
        Initialize model with configuration.

        Args:
            config: Model configuration dictionary
        """
        self.config = config
        self.model = None
        self.is_fitted = False
        self.feature_names = None
        self.metadata = {
            "model_type": self.__class__.__name__,
            "config": config,
            "training_history": []
        }

    @abstractmethod
    def fit(
        self,
        X: Union[np.ndarray, pd.DataFrame],
        y: Union[np.ndarray, pd.Series],
        **kwargs
    ) -> "BaseModel":
        """
        Train the model on provided data.

        Args:
            X: Feature matrix
            y: Target vector
            **kwargs: Additional training arguments

        Returns:
            self: Fitted model instance
        """
        pass

    @abstractmethod
    def predict(
        self,
        X: Union[np.ndarray, pd.DataFrame]
    ) -> np.ndarray:
        """
        Generate predictions for input data.

        Args:
            X: Feature matrix

        Returns:
            Predictions array
        """
        pass

    def predict_proba(
        self,
        X: Union[np.ndarray, pd.DataFrame]
    ) -> np.ndarray:
        """
        Generate probability predictions (for classifiers).

        Args:
            X: Feature matrix

        Returns:
            Probability predictions
        """
        raise NotImplementedError(
            f"{self.__class__.__name__} does not support probability predictions"
        )

    def save(self, path: Union[str, Path]) -> None:
        """
        Save model to disk.

        Args:
            path: Path to save model
        """
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)

        # Save model object
        model_path = path.with_suffix('.pkl')
        joblib.dump(self.model, model_path)

        # Save metadata
        metadata_path = path.with_suffix('.json')
        with open(metadata_path, 'w') as f:
            # Convert non-serializable objects
            serializable_metadata = self._make_serializable(self.metadata)
            json.dump(serializable_metadata, f, indent=2)

        # Save feature names if available
        if self.feature_names is not None:
            features_path = path.with_name(f"{path.stem}_features.json")
            with open(features_path, 'w') as f:
                json.dump(self.feature_names, f)

    @classmethod
    def load(cls, path: Union[str, Path]) -> "BaseModel":
        """
        Load model from disk.

        Args:
            path: Path to saved model

        Returns:
            Loaded model instance
        """
        path = Path(path)

        # Load metadata
        metadata_path = path.with_suffix('.json')
        with open(metadata_path, 'r') as f:
            metadata = json.load(f)

        # Create instance
        instance = cls(metadata['config'])

        # Load model object
        model_path = path.with_suffix('.pkl')
        instance.model = joblib.load(model_path)
        instance.is_fitted = True
        instance.metadata = metadata

        # Load feature names if available
        features_path = path.with_name(f"{path.stem}_features.json")
        if features_path.exists():
            with open(features_path, 'r') as f:
                instance.feature_names = json.load(f)

        return instance

    def get_params(self) -> Dict[str, Any]:
        """Get model parameters."""
        return self.config

    def set_params(self, **params) -> "BaseModel":
        """Set model parameters."""
        self.config.update(params)
        return self

    def _make_serializable(self, obj: Any) -> Any:
        """Convert object to JSON-serializable format."""
        if isinstance(obj, (np.integer, np.floating)):
            return float(obj)
        elif isinstance(obj, np.ndarray):
            return obj.tolist()
        elif isinstance(obj, pd.DataFrame):
            return obj.to_dict()
        elif isinstance(obj, dict):
            return {k: self._make_serializable(v) for k, v in obj.items()}
        elif isinstance(obj, (list, tuple)):
            return [self._make_serializable(item) for item in obj]
        else:
            return obj

    def __repr__(self) -> str:
        params_str = ", ".join(f"{k}={v}" for k, v in self.config.items())
        return f"{self.__class__.__name__}({params_str})"
\end{lstlisting}

# src/models/sklearn_models.py
"""Scikit-learn model implementations."""

from typing import Any, Dict, Union, Optional
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.linear_model import LogisticRegression
from .base import BaseModel


class SklearnModel(BaseModel):
    """
    Wrapper for scikit-learn models with unified interface.
    """

    MODEL_REGISTRY = {
        'random_forest': RandomForestClassifier,
        'gradient_boosting': GradientBoostingClassifier,
        'logistic_regression': LogisticRegression,
    }

    def __init__(self, config: Dict[str, Any]):
        """
        Initialize sklearn model.

        Args:
            config: Must contain 'model_type' key
        """
        super().__init__(config)

        model_type = config.get('model_type')
        if model_type not in self.MODEL_REGISTRY:
            raise ValueError(
                f"Unknown model type: {model_type}. "
                f"Available: {list(self.MODEL_REGISTRY.keys())}"
            )

        # Extract model parameters
        model_params = {
            k: v for k, v in config.items()
            if k not in ['model_type', 'random_state']
        }

        # Initialize model
        model_class = self.MODEL_REGISTRY[model_type]
        self.model = model_class(
            random_state=config.get('random_state', 42),
            **model_params
        )

    def fit(
        self,
        X: Union[np.ndarray, pd.DataFrame],
        y: Union[np.ndarray, pd.Series],
        validation_data: Optional[tuple] = None,
        **kwargs
    ) -> "SklearnModel":
        """
        Fit sklearn model.

        Args:
            X: Training features
            y: Training targets
            validation_data: Optional (X_val, y_val) tuple
            **kwargs: Additional fit parameters

        Returns:
            self: Fitted model
        """
        # Store feature names if DataFrame
        if isinstance(X, pd.DataFrame):
            self.feature_names = X.columns.tolist()

        # Fit model
        self.model.fit(X, y, **kwargs)
        self.is_fitted = True

        # Record training metadata
        self.metadata['training_history'].append({
            'n_samples': len(X),
            'n_features': X.shape[1],
        })

        # Evaluate on validation set if provided
        if validation_data is not None:
            X_val, y_val = validation_data
            val_score = self.model.score(X_val, y_val)
            self.metadata['validation_score'] = val_score

        return self

    def predict(self, X: Union[np.ndarray, pd.DataFrame]) -> np.ndarray:
        """Generate predictions."""
        if not self.is_fitted:
            raise ValueError("Model must be fitted before prediction")

        return self.model.predict(X)

    def predict_proba(self, X: Union[np.ndarray, pd.DataFrame]) -> np.ndarray:
        """Generate probability predictions."""
        if not self.is_fitted:
            raise ValueError("Model must be fitted before prediction")

        if not hasattr(self.model, 'predict_proba'):
            raise NotImplementedError(
                f"{self.config['model_type']} does not support predict_proba"
            )

        return self.model.predict_proba(X)

    def get_feature_importance(self) -> Optional[pd.Series]:
        """
        Get feature importance if available.

        Returns:
            Series with feature importances or None
        """
        if not hasattr(self.model, 'feature_importances_'):
            return None

        importance = self.model.feature_importances_
        if self.feature_names is not None:
            return pd.Series(importance, index=self.feature_names).sort_values(
                ascending=False
            )
        else:
            return pd.Series(importance).sort_values(ascending=False)

# src/models/__init__.py
"""Model factory and registry."""

from typing import Dict, Any, Type
from .base import BaseModel
from .sklearn_models import SklearnModel


class ModelFactory:
    """
    Factory for creating model instances.
    Implements registry pattern for extensibility.
    """

    _registry: Dict[str, Type[BaseModel]] = {
        'sklearn': SklearnModel,
    }

    @classmethod
    def register(cls, name: str, model_class: Type[BaseModel]) -> None:
        """
        Register a new model class.

        Args:
            name: Model identifier
            model_class: Model class to register
        """
        if not issubclass(model_class, BaseModel):
            raise ValueError(f"{model_class} must inherit from BaseModel")

        cls._registry[name] = model_class

    @classmethod
    def create(cls, model_type: str, config: Dict[str, Any]) -> BaseModel:
        """
        Create model instance from configuration.

        Args:
            model_type: Type of model to create
            config: Model configuration

        Returns:
            Model instance

        Example:
            >>> config = {'model_type': 'random_forest', 'n_estimators': 100}
            >>> model = ModelFactory.create('sklearn', config)
        """
        if model_type not in cls._registry:
            raise ValueError(
                f"Unknown model type: {model_type}. "
                f"Available: {list(cls._registry.keys())}"
            )

        model_class = cls._registry[model_type]
        return model_class(config)

    @classmethod
    def available_models(cls) -> list:
        """Get list of registered model types."""
        return list(cls._registry.keys())

\subsection{Data Pipeline Architecture}

\begin{lstlisting}[style=python, caption=Modular data processing pipeline]
# src/data/pipeline.py
"""Data processing pipeline with composable transformations."""

from typing import List, Optional, Dict, Any
from abc import ABC, abstractmethod
import pandas as pd
import numpy as np
from sklearn.base import BaseEstimator, TransformerMixin


class DataTransformer(ABC):
    """Base class for data transformations."""

    def __init__(self, name: str):
        self.name = name
        self.is_fitted = False

    @abstractmethod
    def fit(self, df: pd.DataFrame) -> "DataTransformer":
        """Learn parameters from data."""
        pass

    @abstractmethod
    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Apply transformation to data."""
        pass

    def fit_transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Fit and transform in one step."""
        return self.fit(df).transform(df)


class MissingValueImputer(DataTransformer):
    """Impute missing values with configurable strategy."""

    def __init__(
        self,
        strategy: str = 'mean',
        columns: Optional[List[str]] = None
    ):
        super().__init__(name='MissingValueImputer')
        self.strategy = strategy
        self.columns = columns
        self.fill_values = {}

    def fit(self, df: pd.DataFrame) -> "MissingValueImputer":
        """Learn imputation values from data."""
        cols = self.columns or df.select_dtypes(include=[np.number]).columns

        for col in cols:
            if self.strategy == 'mean':
                self.fill_values[col] = df[col].mean()
            elif self.strategy == 'median':
                self.fill_values[col] = df[col].median()
            elif self.strategy == 'mode':
                self.fill_values[col] = df[col].mode()[0]
            elif self.strategy == 'zero':
                self.fill_values[col] = 0
            else:
                raise ValueError(f"Unknown strategy: {self.strategy}")

        self.is_fitted = True
        return self

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Apply imputation."""
        if not self.is_fitted:
            raise ValueError("Transformer must be fitted before transform")

        df = df.copy()
        for col, value in self.fill_values.items():
            if col in df.columns:
                df[col] = df[col].fillna(value)

        return df


class OutlierClipper(DataTransformer):
    """Clip outliers based on percentiles."""

    def __init__(
        self,
        lower_percentile: float = 1,
        upper_percentile: float = 99,
        columns: Optional[List[str]] = None
    ):
        super().__init__(name='OutlierClipper')
        self.lower_percentile = lower_percentile
        self.upper_percentile = upper_percentile
        self.columns = columns
        self.clip_values = {}

    def fit(self, df: pd.DataFrame) -> "OutlierClipper":
        """Learn clipping thresholds."""
        cols = self.columns or df.select_dtypes(include=[np.number]).columns

        for col in cols:
            lower = df[col].quantile(self.lower_percentile / 100)
            upper = df[col].quantile(self.upper_percentile / 100)
            self.clip_values[col] = (lower, upper)

        self.is_fitted = True
        return self

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Apply clipping."""
        if not self.is_fitted:
            raise ValueError("Transformer must be fitted before transform")

        df = df.copy()
        for col, (lower, upper) in self.clip_values.items():
            if col in df.columns:
                df[col] = df[col].clip(lower=lower, upper=upper)

        return df


class DataPipeline:
    """
    Composable data processing pipeline.
    Chains multiple transformers together.
    """

    def __init__(self, transformers: List[DataTransformer]):
        """
        Initialize pipeline with transformers.

        Args:
            transformers: List of DataTransformer instances
        """
        self.transformers = transformers
        self.is_fitted = False

    def fit(self, df: pd.DataFrame) -> "DataPipeline":
        """Fit all transformers sequentially."""
        data = df.copy()

        for transformer in self.transformers:
            transformer.fit(data)
            data = transformer.transform(data)

        self.is_fitted = True
        return self

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Apply all transformers sequentially."""
        if not self.is_fitted:
            raise ValueError("Pipeline must be fitted before transform")

        data = df.copy()
        for transformer in self.transformers:
            data = transformer.transform(data)

        return data

    def fit_transform(self, df: pd.DataFrame) -> pd.DataFrame:
        """Fit and transform in one step."""
        return self.fit(df).transform(df)

    def add_transformer(self, transformer: DataTransformer) -> "DataPipeline":
        """Add transformer to pipeline."""
        self.transformers.append(transformer)
        self.is_fitted = False  # Need to refit
        return self


# Example usage
if __name__ == "__main__":

from abc import ABC, abstractmethod
from typing import Dict, Any, Tuple
import pandas as pd
import numpy as np
from sklearn.model_selection import GridSearchCV, RandomizedSearchCV

class TrainingStrategy(ABC):
    """Abstract base class for training strategies."""

    @abstractmethod
    def train(
        self,
        model: BaseModel,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> BaseModel:
        """Execute training strategy."""
        pass


class SimpleTrainingStrategy(TrainingStrategy):
    """Simple training without hyperparameter optimization."""

    def train(
        self,
        model: BaseModel,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> BaseModel:
        """Train model with default parameters."""
        model.fit(X_train, y_train, validation_data=(X_val, y_val))
        return model


class GridSearchStrategy(TrainingStrategy):
    """Grid search hyperparameter optimization."""

    def __init__(self, param_grid: Dict[str, list], cv: int = 5):
        """
        Initialize grid search strategy.

        Args:
            param_grid: Parameter grid for search
            cv: Number of cross-validation folds
        """
        self.param_grid = param_grid
        self.cv = cv

    def train(
        self,
        model: BaseModel,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> BaseModel:
        """Train with grid search."""
        # Use sklearn's GridSearchCV on underlying model
        grid_search = GridSearchCV(
            model.model,
            self.param_grid,
            cv=self.cv,
            scoring='accuracy',
            n_jobs=-1
        )

        grid_search.fit(X_train, y_train)

        # Update model with best parameters
        model.model = grid_search.best_estimator_
        model.is_fitted = True
        model.metadata['best_params'] = grid_search.best_params_
        model.metadata['best_score'] = grid_search.best_score_

        return model


class EarlyStoppingStrategy(TrainingStrategy):
    """Training with early stopping (for iterative models)."""

    def __init__(
        self,
        patience: int = 5,
        min_delta: float = 0.001
    ):
        """
        Initialize early stopping strategy.

        Args:
            patience: Number of epochs to wait for improvement
            min_delta: Minimum change to qualify as improvement
        """
        self.patience = patience
        self.min_delta = min_delta

    def train(
        self,
        model: BaseModel,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> BaseModel:
        """Train with early stopping logic."""
        # For sklearn models with iterative training
        if hasattr(model.model, 'n_estimators'):
            best_score = -np.inf
            patience_counter = 0

            for n_est in range(10, 201, 10):
                model.model.set_params(n_estimators=n_est)
                model.fit(X_train, y_train)

                val_score = model.model.score(X_val, y_val)

                if val_score > best_score + self.min_delta:
                    best_score = val_score
                    patience_counter = 0
                else:
                    patience_counter += 1

                if patience_counter >= self.patience:
                    print(f"Early stopping at {n_est} estimators")
                    break

            model.metadata['early_stopped_at'] = n_est
            model.metadata['best_val_score'] = best_score

        else:
            # Fallback to simple training
            model.fit(X_train, y_train, validation_data=(X_val, y_val))

        return model


class TrainingContext:
    """Context that uses a training strategy."""

    def __init__(self, strategy: TrainingStrategy):
        """
        Initialize with specific strategy.

        Args:
            strategy: Training strategy to use
        """
        self.strategy = strategy

    def set_strategy(self, strategy: TrainingStrategy):
        """Change strategy at runtime."""
        self.strategy = strategy

    def execute_training(
        self,
        model: BaseModel,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> BaseModel:
        """Execute training using current strategy."""
        return self.strategy.train(model, X_train, y_train, X_val, y_val)

# src/training/callbacks.py
"""Training callbacks using Observer pattern."""

from abc import ABC, abstractmethod
from typing import Dict, Any, List
import time
from datetime import datetime


class TrainingCallback(ABC):
    """Abstract base class for training callbacks."""

    @abstractmethod
    def on_train_begin(self, logs: Dict[str, Any] = None):
        """Called at the beginning of training."""
        pass

    @abstractmethod
    def on_train_end(self, logs: Dict[str, Any] = None):
        """Called at the end of training."""
        pass

    @abstractmethod
    def on_epoch_begin(self, epoch: int, logs: Dict[str, Any] = None):
        """Called at the beginning of an epoch."""
        pass

    @abstractmethod
    def on_epoch_end(self, epoch: int, logs: Dict[str, Any] = None):
        """Called at the end of an epoch."""
        pass


class LoggingCallback(TrainingCallback):
    """Log training progress."""

    def on_train_begin(self, logs: Dict[str, Any] = None):
        """Log training start."""
        print(f"[{datetime.now()}] Training started")
        print(f"Configuration: {logs.get('config', {})}")

    def on_train_end(self, logs: Dict[str, Any] = None):
        """Log training completion."""
        duration = logs.get('duration', 0)
        print(f"[{datetime.now()}] Training completed in {duration:.2f}s")

    def on_epoch_begin(self, epoch: int, logs: Dict[str, Any] = None):
        """Log epoch start."""
        print(f"\nEpoch {epoch}/{logs.get('total_epochs', '?')} started")

    def on_epoch_end(self, epoch: int, logs: Dict[str, Any] = None):
        """Log epoch metrics."""
        metrics_str = ", ".join(
            f"{k}={v:.4f}" for k, v in logs.items()
            if k not in ['total_epochs']
        )
        print(f"Epoch {epoch} - {metrics_str}")


class MetricTrackingCallback(TrainingCallback):
    """Track metrics across epochs."""

    def __init__(self):
        self.history = {
            'epoch': [],
            'train_loss': [],
            'val_loss': [],
            'train_acc': [],
            'val_acc': []
        }

    def on_train_begin(self, logs: Dict[str, Any] = None):
        """Reset history."""
        self.history = {k: [] for k in self.history.keys()}

    def on_train_end(self, logs: Dict[str, Any] = None):
        """Store final history."""
        pass

    def on_epoch_begin(self, epoch: int, logs: Dict[str, Any] = None):
        """No action needed."""
        pass

    def on_epoch_end(self, epoch: int, logs: Dict[str, Any] = None):
        """Record metrics."""
        self.history['epoch'].append(epoch)
        for key in ['train_loss', 'val_loss', 'train_acc', 'val_acc']:
            value = logs.get(key, None)
            self.history[key].append(value)


class EarlyStoppingCallback(TrainingCallback):
    """Stop training when metric stops improving."""

    def __init__(
        self,
        monitor: str = 'val_loss',
        patience: int = 5,
        min_delta: float = 0.001,
        mode: str = 'min'
    ):
        """
        Initialize early stopping.

        Args:
            monitor: Metric to monitor
            patience: Epochs to wait for improvement
            min_delta: Minimum change to qualify as improvement
            mode: 'min' or 'max' (whether lower or higher is better)
        """
        self.monitor = monitor
        self.patience = patience
        self.min_delta = min_delta
        self.mode = mode

        self.best_value = float('inf') if mode == 'min' else float('-inf')
        self.wait = 0
        self.stop_training = False

    def on_train_begin(self, logs: Dict[str, Any] = None):
        """Reset state."""
        self.best_value = float('inf') if self.mode == 'min' else float('-inf')
        self.wait = 0
        self.stop_training = False

    def on_train_end(self, logs: Dict[str, Any] = None):
        """No action needed."""
        pass

    def on_epoch_begin(self, epoch: int, logs: Dict[str, Any] = None):
        """No action needed."""
        pass

    def on_epoch_end(self, epoch: int, logs: Dict[str, Any] = None):
        """Check if should stop."""
        current_value = logs.get(self.monitor)
        if current_value is None:
            return

        if self.mode == 'min':
            improved = current_value < (self.best_value - self.min_delta)
        else:
            improved = current_value > (self.best_value + self.min_delta)

        if improved:
            self.best_value = current_value
            self.wait = 0
        else:
            self.wait += 1
            if self.wait >= self.patience:
                self.stop_training = True
                print(f"\nEarly stopping triggered at epoch {epoch}")


class CallbackList:
    """Manages multiple callbacks."""

    def __init__(self, callbacks: List[TrainingCallback] = None):
        """Initialize with list of callbacks."""
        self.callbacks = callbacks or []

    def append(self, callback: TrainingCallback):
        """Add callback to list."""
        self.callbacks.append(callback)

    def on_train_begin(self, logs: Dict[str, Any] = None):
        """Notify all callbacks."""
        logs = logs or {}
        for callback in self.callbacks:
            callback.on_train_begin(logs)

    def on_train_end(self, logs: Dict[str, Any] = None):
        """Notify all callbacks."""
        logs = logs or {}
        for callback in self.callbacks:
            callback.on_train_end(logs)

    def on_epoch_begin(self, epoch: int, logs: Dict[str, Any] = None):
        """Notify all callbacks."""
        logs = logs or {}
        for callback in self.callbacks:
            callback.on_epoch_begin(epoch, logs)

    def on_epoch_end(self, epoch: int, logs: Dict[str, Any] = None):
        """Notify all callbacks."""
        logs = logs or {}
        for callback in self.callbacks:
            callback.on_epoch_end(epoch, logs)

    def should_stop(self) -> bool:
        """Check if any callback requested stop."""
        return any(
            getattr(cb, 'stop_training', False)
            for cb in self.callbacks
        )

def set_seed(seed: int) -> None:
    """
    Set random seeds for reproducibility.

    Args:
        seed: Random seed value
    """
    random.seed(seed)
    np.random.seed(seed)
    os.environ['PYTHONHASHSEED'] = str(seed)

    # For frameworks like TensorFlow/PyTorch
    try:
        import torch
        torch.manual_seed(seed)
        torch.cuda.manual_seed_all(seed)
        torch.backends.cudnn.deterministic = True
        torch.backends.cudnn.benchmark = False
    except ImportError:
        pass

    try:
        import tensorflow as tf
        tf.random.set_seed(seed)
    except ImportError:
        pass


def load_data(config: ExperimentConfig) -> tuple:
    """
    Load and prepare data.

    Args:
        config: Experiment configuration

    Returns:
        Tuple of (X_train, y_train, X_val, y_val)
    """
    logger.info("Loading data...")

    # Load datasets
    train_df = pd.read_csv(config.data.train_path)
    val_df = pd.read_csv(config.data.val_path)

    # Extract features and target
    target_col = config.data.target

    if config.data.features:
        feature_cols = config.data.features
    else:
        feature_cols = [col for col in train_df.columns if col != target_col]

    X_train = train_df[feature_cols]
    y_train = train_df[target_col]
    X_val = val_df[feature_cols]
    y_val = val_df[target_col]

    logger.info(f"Training set: {X_train.shape}")
    logger.info(f"Validation set: {X_val.shape}")

    return X_train, y_train, X_val, y_val


def create_preprocessing_pipeline() -> DataPipeline:
    """Create data preprocessing pipeline."""
    return DataPipeline([
        MissingValueImputer(strategy='median')
    ])


def train(config_path: Path, experiment: bool = False) -> None:
    """
    Execute reproducible training.

    Args:
        config_path: Path to configuration file
        experiment: Whether this is an experiment run
    """
    # Load configuration
    logger.info(f"Loading configuration from {config_path}")
    config = ExperimentConfig.from_yaml(config_path)

    # Set seeds for reproducibility
    logger.info(f"Setting random seed: {config.seed}")
    set_seed(config.seed)

    # Create experiment directory
    exp_dir = Path("experiments") / config.name
    exp_dir.mkdir(parents=True, exist_ok=True)

    # Save configuration to experiment directory
    config.to_yaml(exp_dir / "config.yaml")
    config.to_json(exp_dir / "config.json")

    logger.info(f"Experiment: {config.name}")
    logger.info(f"Config hash: {config.get_hash()}")

    # Load data
    X_train, y_train, X_val, y_val = load_data(config)

    # Create preprocessing pipeline
    logger.info("Creating preprocessing pipeline...")
    preprocess_pipeline = create_preprocessing_pipeline()

    # Fit and transform data
    X_train = preprocess_pipeline.fit_transform(X_train)
    X_val = preprocess_pipeline.transform(X_val)

    # Create model
    logger.info("Creating model...")
    model = ModelFactory.create(
        config.model.model_type,
        {
            'model_type': config.model.model_class,
            **config.model.parameters,
            'random_state': config.model.random_state
        }
    )

    # Create callbacks
    callbacks = CallbackList([
        LoggingCallback(),
        MetricTrackingCallback(),
        EarlyStoppingCallback(patience=config.training.early_stopping_patience)
    ])

    # Train model
    logger.info("Training model...")
    callbacks.on_train_begin({'config': config.to_dict()})

    start_time = pd.Timestamp.now()
    model.fit(X_train, y_train, validation_data=(X_val, y_val))
    end_time = pd.Timestamp.now()

    training_duration = (end_time - start_time).total_seconds()

    callbacks.on_train_end({'duration': training_duration})

    # Evaluate model
    logger.info("Evaluating model...")
    train_score = model.model.score(X_train, y_train)
    val_score = model.model.score(X_val, y_val)

    logger.info(f"Training accuracy: {train_score:.4f}")
    logger.info(f"Validation accuracy: {val_score:.4f}")

    # Save model
    model_path = exp_dir / "model"
    logger.info(f"Saving model to {model_path}")
    model.save(model_path)

    # Save metrics
    metrics = {
        'train_accuracy': float(train_score),
        'val_accuracy': float(val_score),
        'training_duration_seconds': training_duration,
        'config_hash': config.get_hash()
    }

    with open(exp_dir / "metrics.json", 'w') as f:
        json.dump(metrics, f, indent=2)

    logger.info("Training completed successfully!")

# src/utils/versioning.py
"""Model and experiment versioning utilities."""

from typing import Dict, Any, Optional, List
from pathlib import Path
from datetime import datetime
import json
import hashlib
from dataclasses import dataclass, asdict


@dataclass
class ModelVersion:
    """Represents a versioned model."""
    version_id: str
    model_path: Path
    config_hash: str
    created_at: str
    metrics: Dict[str, float]
    parent_version: Optional[str] = None
    tags: Optional[List[str]] = None
    notes: Optional[str] = None


class ModelRegistry:
    """
    Central registry for tracking model versions.
    Enables model lineage tracking and rollback.
    """

    def __init__(self, registry_path: Path):
        """
        Initialize model registry.

        Args:
            registry_path: Path to registry storage
        """
        self.registry_path = Path(registry_path)
        self.registry_path.mkdir(parents=True, exist_ok=True)
        self.index_file = self.registry_path / "index.json"
        self.versions: Dict[str, ModelVersion] = {}
        self._load_index()

    def _load_index(self):
        """Load registry index from disk."""
        if self.index_file.exists():
            with open(self.index_file, 'r') as f:
                data = json.load(f)
                self.versions = {
                    k: ModelVersion(**v) for k, v in data.items()
                }

    def _save_index(self):
        """Save registry index to disk."""
        data = {k: asdict(v) for k, v in self.versions.items()}
        # Convert Path objects to strings
        for v in data.values():
            v['model_path'] = str(v['model_path'])

        with open(self.index_file, 'w') as f:
            json.dump(data, f, indent=2)

    def register_model(
        self,
        model_path: Path,
        config: Dict[str, Any],
        metrics: Dict[str, float],
        parent_version: Optional[str] = None,
        tags: Optional[List[str]] = None,
        notes: Optional[str] = None
    ) -> str:
        """
        Register a new model version.

        Args:
            model_path: Path to saved model
            config: Model configuration
            metrics: Performance metrics
            parent_version: ID of parent version (for lineage)
            tags: Optional tags for categorization
            notes: Optional notes about this version

        Returns:
            version_id: Unique version identifier
        """
        # Generate version ID
        timestamp = datetime.now().isoformat()
        config_str = json.dumps(config, sort_keys=True)
        config_hash = hashlib.md5(config_str.encode()).hexdigest()[:8]
        version_id = f"v_{datetime.now().strftime('%Y%m%d_%H%M%S')}_{config_hash}"

        # Create version record
        version = ModelVersion(
            version_id=version_id,
            model_path=Path(model_path),
            config_hash=config_hash,
            created_at=timestamp,
            metrics=metrics,
            parent_version=parent_version,
            tags=tags or [],
            notes=notes
        )

        self.versions[version_id] = version
        self._save_index()

        return version_id

    def get_version(self, version_id: str) -> Optional[ModelVersion]:
        """Get specific model version."""
        return self.versions.get(version_id)

    def get_best_version(
        self,
        metric: str,
        higher_is_better: bool = True
    ) -> Optional[ModelVersion]:
        """
        Get best model version by metric.

        Args:
            metric: Metric name to compare
            higher_is_better: Whether higher metric values are better

        Returns:
            Best model version or None
        """
        valid_versions = [
            v for v in self.versions.values()
            if metric in v.metrics
        ]

        if not valid_versions:
            return None

        return max(
            valid_versions,
            key=lambda v: v.metrics[metric] if higher_is_better
            else -v.metrics[metric]
        )

    def get_lineage(self, version_id: str) -> List[ModelVersion]:
        """
        Get complete lineage of a model version.

        Args:
            version_id: Version to trace

        Returns:
            List of versions from root to specified version
        """
        lineage = []
        current_id = version_id

        while current_id is not None:
            version = self.versions.get(current_id)
            if version is None:
                break
            lineage.append(version)
            current_id = version.parent_version

        return list(reversed(lineage))

    def list_versions(
        self,
        tags: Optional[List[str]] = None,
        min_metric: Optional[Dict[str, float]] = None
    ) -> List[ModelVersion]:
        """
        List model versions with optional filtering.

        Args:
            tags: Filter by tags
            min_metric: Filter by minimum metric values

        Returns:
            List of matching model versions
        """
        versions = list(self.versions.values())

        # Filter by tags
        if tags:
            versions = [
                v for v in versions
                if any(tag in (v.tags or []) for tag in tags)
            ]

        # Filter by metrics
        if min_metric:
            for metric, min_value in min_metric.items():
                versions = [
                    v for v in versions
                    if metric in v.metrics and v.metrics[metric] >= min_value
                ]

        return sorted(versions, key=lambda v: v.created_at, reverse=True)

            for bias in self.potential_biases:

            md += "\n"

"""
Example of well-documented ML code with comprehensive docstrings.
Follows Google docstring style.

from typing import Union, Optional, Tuple, List
import numpy as np
import pandas as pd
from numpy.typing import NDArray


class DocumentedModel:
    """
    Production-ready ML model with comprehensive documentation.

    This model provides a unified interface for training and inference
    with automatic feature validation and preprocessing.

    Attributes:
        config: Model configuration dictionary
        is_fitted: Whether model has been trained
        feature_names_: List of feature names from training
        classes_: Unique class labels (for classification)
        n_features_in_: Number of features seen during fit

    Example:
        Basic usage::

            >>> from sklearn.datasets import make_classification
            >>> X, y = make_classification(n_samples=1000, random_state=42)
            >>> model = DocumentedModel({'max_depth': 5})
            >>> model.fit(X, y)
            >>> predictions = model.predict(X[:10])
            >>> print(predictions.shape)
            (10,)

        With validation data::

            >>> X_train, X_val = X[:800], X[800:]
            >>> y_train, y_val = y[:800], y[800:]
            >>> model.fit(X_train, y_train, validation_data=(X_val, y_val))
            >>> print(model.metadata['validation_score'])
            0.85

    Note:
        This model requires features to be preprocessed (scaled, encoded)
        before training or inference.

    .. versionadded:: 1.0.0
    .. versionchanged:: 1.2.0
       Added support for sample weights in fit method
    """

    def __init__(self, config: dict):
        """
        Initialize model with configuration.

        Args:
            config: Configuration dictionary with model parameters.
                   Required keys depend on model_type.

        Raises:
            ValueError: If required configuration keys are missing

        Example:
            >>> config = {
            ...     'model_type': 'random_forest',
            ...     'n_estimators': 100,
            ...     'max_depth': 10
            ... }
            >>> model = DocumentedModel(config)
        """
        self.config = config
        self.is_fitted = False
        self.feature_names_: Optional[List[str]] = None
        self.classes_: Optional[NDArray] = None
        self.n_features_in_: Optional[int] = None
        self.metadata = {}

    def fit(
        self,
        X: Union[NDArray[np.float64], pd.DataFrame],
        y: Union[NDArray[np.int_], pd.Series],
        validation_data: Optional[Tuple] = None,
        sample_weight: Optional[NDArray[np.float64]] = None
    ) -> "DocumentedModel":
        """
        Fit model to training data.

        Trains the model on provided features and targets. Optionally
        evaluates on validation set and applies sample weights.

        Args:
            X: Training features of shape (n_samples, n_features).
               Can be numpy array or pandas DataFrame.
            y: Training targets of shape (n_samples,).
               For classification: integer class labels.
               For regression: continuous values.
            validation_data: Optional tuple of (X_val, y_val) for
                           validation during training. If provided,
                           validation metrics are stored in metadata.
            sample_weight: Optional array of shape (n_samples,) with
                         sample weights. Higher weights increase the
                         importance of the corresponding samples.

        Returns:
            self: Fitted model instance (enables method chaining)

        Raises:
            ValueError: If X and y have incompatible shapes
            ValueError: If sample_weight has wrong shape
            RuntimeError: If model fitting fails

        Example:
            >>> X_train = np.random.randn(100, 5)
            >>> y_train = np.random.randint(0, 2, 100)
            >>> model = DocumentedModel({'model_type': 'logistic_regression'})
            >>> model.fit(X_train, y_train)
            DocumentedModel(...)

            With validation::

                >>> X_val = np.random.randn(20, 5)
                >>> y_val = np.random.randint(0, 2, 20)
                >>> model.fit(X_train, y_train, validation_data=(X_val, y_val))
                DocumentedModel(...)

        Note:
            After fitting, the following attributes are set:
            - is_fitted: True
            - feature_names_: Feature names (if X is DataFrame)
            - n_features_in_: Number of features
            - classes_: Unique classes (for classification)

        .. versionadded:: 1.0.0
        .. versionchanged:: 1.2.0
           Added sample_weight parameter
        """
        # Validate inputs
        if len(X) != len(y):
            raise ValueError(
                f"X and y have incompatible shapes: {len(X)} vs {len(y)}"
            )

        if sample_weight is not None and len(sample_weight) != len(X):
            raise ValueError(
                f"sample_weight has wrong shape: {len(sample_weight)} vs {len(X)}"
            )

        # Extract feature names if DataFrame
        if isinstance(X, pd.DataFrame):
            self.feature_names_ = X.columns.tolist()

        # Store training metadata
        self.n_features_in_ = X.shape[1]
        self.classes_ = np.unique(y)
        self.is_fitted = True

        # Implementation details...
        return self

    def predict(
        self,
        X: Union[NDArray[np.float64], pd.DataFrame]
    ) -> NDArray[np.int_]:
        """
        Generate predictions for input features.

        Args:
            X: Features of shape (n_samples, n_features).
               Must have same number of features as training data.

        Returns:
            Predictions of shape (n_samples,).
            For classification: predicted class labels.
            For regression: predicted values.

        Raises:
            ValueError: If model is not fitted
            ValueError: If X has wrong number of features

        Example:
            >>> model.fit(X_train, y_train)
            >>> predictions = model.predict(X_test)
            >>> print(predictions[:5])
            [0 1 1 0 1]

        Warning:
            Input features must be preprocessed the same way as
            training data. Inconsistent preprocessing will lead
            to degraded performance.

        See Also:
            predict_proba: Generate probability predictions
            predict_log_proba: Generate log probability predictions

        .. versionadded:: 1.0.0
        """
        if not self.is_fitted:
            raise ValueError(
                "Model must be fitted before calling predict(). "
                "Call fit(X, y) first."
            )

        if X.shape[1] != self.n_features_in_:
            raise ValueError(
                f"X has {X.shape[1]} features but model expects "
                f"{self.n_features_in_} features"
            )

        # Implementation details...
        return np.zeros(len(X), dtype=np.int_)

      - name: Notify deployment
        run: |
          echo "Model deployed to production"

\section{Chapter Summary}

This chapter presented comprehensive best practices for professional ML development, bridging the gap between research code and production systems. Key takeaways include:

\begin{itemize}
    \item \textbf{Project Structure:} Well-organized projects with clear separation of concerns enable collaboration and maintainability at scale.

    \item \textbf{Modular Architecture:} Object-oriented design with base classes, factories, and composable components creates flexible, testable systems.

    \item \textbf{Comprehensive Testing:} Multi-level testing strategies (unit, integration, model-specific) ensure reliability and catch regressions early.

    \item \textbf{Design Patterns:} Adapting software design patterns to ML workflows (Strategy, Observer, Factory) provides proven solutions to common problems.

    \item \textbf{Code Quality:} Automated tools (Black, isort, flake8, mypy) and pre-commit hooks maintain consistent, high-quality code.

    \item \textbf{Reproducibility:} Configuration management, seed setting, and experiment tracking enable reproducible research and debugging.

    \item \textbf{Type Safety:} Type hints improve code documentation, enable static analysis, and catch bugs before runtime.

    \item \textbf{Automation:} Makefiles, CI/CD pipelines, and automated testing reduce manual work and ensure consistency.
\end{itemize}

The transition from experimental to production ML code requires discipline and structure. Organizations that invest in these practices report significantly improved development velocity, reduced bugs, and easier collaboration across teams. The patterns and tools presented in this chapter form the foundation for building reliable, maintainable ML systems at scale.
