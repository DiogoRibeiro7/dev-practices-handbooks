\section{Introduction: Model Serving Architecture}

Model serving is the process of deploying machine learning models to production where they can accept requests and return predictions. Unlike training, which is typically a batch process, serving must handle:

\begin{itemize}
    \item \textbf{Real-time requirements}: Predictions often needed in milliseconds
    \item \textbf{High availability}: 99.9\%+ uptime requirements
    \item \textbf{Variable load}: Traffic can spike unpredictably
    \item \textbf{Cost efficiency}: Minimize infrastructure costs at scale
    \item \textbf{Multiple models}: Version management and A/B testing
\end{itemize}

A production serving system typically includes:

\begin{enumerate}
    \item \textbf{API Layer}: REST or gRPC endpoints for client requests
    \item \textbf{Model Server}: Loads models and executes inference
    \item \textbf{Preprocessing}: Feature transformation and validation
    \item \textbf{Postprocessing}: Result formatting and business logic
    \item \textbf{Caching}: Reduce redundant computations
    \item \textbf{Load Balancer}: Distribute traffic across instances
    \item \textbf{Monitoring}: Track latency, throughput, and errors
\end{enumerate}

# Model wrapper with caching and lazy loading
class ModelManager:
    """
    Manage model lifecycle: loading, caching, and inference.

    Features:
    - Lazy loading (load on first request)
    - Model versioning
    - GPU/CPU support
    - Warmup for consistent latency
    """

    def __init__(self, model_path: str, device: str = "cuda"):
        self.model_path = model_path
        self.device = device if torch.cuda.is_available() else "cpu"
        self.model = None
        self.model_version = "v1"
        self.load_time = None

    def load_model(self):
        """Load model from disk."""
        if self.model is not None:
            return

        logger.info(f"Loading model from {self.model_path}")
        start_time = time.time()

        try:
            # Load model checkpoint
            checkpoint = torch.load(self.model_path, map_location=self.device)

            # Reconstruct model
            self.model = self._create_model_architecture(checkpoint)
            self.model.load_state_dict(checkpoint['model_state_dict'])
            self.model.to(self.device)
            self.model.eval()

            self.load_time = time.time() - start_time
            logger.info(f"Model loaded in {self.load_time:.2f}s on {self.device}")

            # Warmup inference
            self._warmup()

        except Exception as e:
            logger.error(f"Failed to load model: {e}")
            raise

    def _create_model_architecture(self, checkpoint: Dict) -> nn.Module:
        """Create model architecture from checkpoint."""
        # Example: Simple feedforward network
        input_dim = checkpoint.get('input_dim', 784)
        hidden_dim = checkpoint.get('hidden_dim', 256)
        output_dim = checkpoint.get('output_dim', 10)

        model = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim)
        )
        return model

    def _warmup(self, n_iterations: int = 10):
        """Warmup model with dummy inputs."""
        logger.info("Warming up model...")
        dummy_input = torch.randn(1, 784).to(self.device)

        with torch.no_grad():
            for _ in range(n_iterations):
                _ = self.model(dummy_input)

        logger.info("Warmup complete")

    @torch.no_grad()
    def predict(self, features: np.ndarray) -> Dict:
        """
        Run inference on input features.

        Args:
            features: Input features (numpy array)

        Returns:
            Dict with prediction, probabilities, and metadata
        """
        if self.model is None:
            self.load_model()

        start_time = time.time()

        # Convert to tensor
        x = torch.from_numpy(features).float().unsqueeze(0).to(self.device)

        # Inference
        logits = self.model(x)
        probabilities = torch.softmax(logits, dim=1)
        prediction = torch.argmax(probabilities, dim=1)

        latency_ms = (time.time() - start_time) * 1000

        return {
            'prediction': prediction.item(),
            'probabilities': probabilities.cpu().numpy().tolist()[0],
            'latency_ms': latency_ms
        }

    @torch.no_grad()
    def predict_batch(self, batch: np.ndarray) -> Dict:
        """Run batch inference."""
        if self.model is None:
            self.load_model()

        start_time = time.time()

        # Convert to tensor
        x = torch.from_numpy(batch).float().to(self.device)

        # Batch inference
        logits = self.model(x)
        predictions = torch.argmax(logits, dim=1)

        latency_ms = (time.time() - start_time) * 1000

        return {
            'predictions': predictions.cpu().numpy().tolist(),
            'latency_ms': latency_ms
        }

class MetricsTracker:

    def __init__(self):
        self.request_count = 0
        self.total_latency = 0.0

        return {
            'total_requests': self.request_count,
            'avg_latency_ms': avg_latency,
            'error_count': self.error_count,

        }

# Redis cache for predictions
class PredictionCache:
    """
    Cache predictions to avoid redundant inference.

    Use cases:
    - Repeated requests for same input
    - Expensive model inference
    - High QPS scenarios
    """

    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis_client = None
        self.redis_url = redis_url
        self.ttl = 3600  # Cache TTL in seconds

    async def connect(self):
        """Connect to Redis."""
        self.redis_client = await redis.from_url(
            self.redis_url,
            encoding="utf-8",
            decode_responses=True
        )

    async def close(self):
        """Close Redis connection."""
        if self.redis_client:
            await self.redis_client.close()

    def _generate_cache_key(self, features: List[float], model_version: str) -> str:
        """Generate cache key from input features."""
        # Hash features for cache key
        features_str = json.dumps(features, sort_keys=True)
        feature_hash = hashlib.md5(features_str.encode()).hexdigest()
        return f"prediction:{model_version}:{feature_hash}"

    async def get(self, features: List[float], model_version: str) -> Optional[Dict]:
        """Get cached prediction."""
        if not self.redis_client:
            return None

        cache_key = self._generate_cache_key(features, model_version)
        cached = await self.redis_client.get(cache_key)

        if cached:
            return json.loads(cached)

        return None

    async def set(self, features: List[float], model_version: str, result: Dict):
        """Cache prediction result."""
        if not self.redis_client:
            return

        cache_key = self._generate_cache_key(features, model_version)
        await self.redis_client.setex(
            cache_key,
            self.ttl,
            json.dumps(result)
        )

class RequestValidator:
    """Validate and preprocess requests."""

    @staticmethod
    def validate_features(features: List[float], expected_dim: int = 784) -> np.ndarray:
        """Validate feature dimensions and values."""
        if len(features) != expected_dim:
            raise HTTPException(
                status_code=400,
                detail=f"Expected {expected_dim} features, got {len(features)}"
            )

        # Check for invalid values
        features_array = np.array(features, dtype=np.float32)

        if np.any(np.isnan(features_array)):
            raise HTTPException(status_code=400, detail="Features contain NaN values")

        if np.any(np.isinf(features_array)):
            raise HTTPException(status_code=400, detail="Features contain Inf values")

        return features_array

    @staticmethod
    def normalize_features(features: np.ndarray) -> np.ndarray:
        """Normalize features to [0, 1] range."""
        min_val = features.min()
        max_val = features.max()

        if max_val - min_val < 1e-8:
            return features

        return (features - min_val) / (max_val - min_val)

import grpc
from concurrent import futures
import time
import numpy as np
import torch
import logging

# Generated from prediction.proto
import prediction_pb2
import prediction_pb2_grpc

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class PredictionServicer(prediction_pb2_grpc.PredictionServiceServicer):
    """
    gRPC service for model predictions.

    Features:
    - High-performance binary protocol
    - Streaming support
    - Built-in load balancing
    - Type-safe requests/responses
    """

    def __init__(self, model_manager: ModelManager):
        self.model_manager = model_manager
        self.model_manager.load_model()

    def Predict(self, request, context):
        """Handle single prediction request."""
        start_time = time.time()

        try:
            # Extract features
            features = np.array(request.features, dtype=np.float32)

            # Run inference
            result = self.model_manager.predict(features)

            latency_ms = (time.time() - start_time) * 1000

            # Build response
            response = prediction_pb2.PredictResponse(
                prediction=int(result['prediction']),
                probabilities=result['probabilities'],
                model_version=request.model_version,
                latency_ms=latency_ms
            )

            return response

        except Exception as e:
            logger.error(f"Prediction error: {e}")
            context.set_code(grpc.StatusCode.INTERNAL)
            context.set_details(str(e))
            return prediction_pb2.PredictResponse()

    def PredictBatch(self, request, context):
        """Handle batch prediction request."""
        start_time = time.time()

        try:
            # Extract batch
            batch = np.array([list(f.values) for f in request.batch], dtype=np.float32)

            # Run batch inference
            result = self.model_manager.predict_batch(batch)

            latency_ms = (time.time() - start_time) * 1000

            response = prediction_pb2.BatchPredictResponse(
                predictions=result['predictions'],
                model_version=request.model_version,
                batch_size=len(request.batch),
                latency_ms=latency_ms
            )

            return response

        except Exception as e:
            logger.error(f"Batch prediction error: {e}")
            context.set_code(grpc.StatusCode.INTERNAL)
            context.set_details(str(e))
            return prediction_pb2.BatchPredictResponse()

    def PredictStream(self, request_iterator, context):
        """Handle streaming predictions."""
        for request in request_iterator:
            try:
                features = np.array(request.features, dtype=np.float32)
                result = self.model_manager.predict(features)

                response = prediction_pb2.PredictResponse(
                    prediction=int(result['prediction']),
                    probabilities=result['probabilities'],
                    model_version=request.model_version,
                    latency_ms=result['latency_ms']
                )

                yield response

            except Exception as e:
                logger.error(f"Streaming prediction error: {e}")
                context.set_code(grpc.StatusCode.INTERNAL)
                context.set_details(str(e))
                return

    def HealthCheck(self, request, context):
        """Health check endpoint."""
        if self.model_manager.model is not None:
            status = prediction_pb2.HealthCheckResponse.SERVING
        else:
            status = prediction_pb2.HealthCheckResponse.NOT_SERVING

        return prediction_pb2.HealthCheckResponse(status=status)

def serve(model_path: str, port: int = 50051, max_workers: int = 10):
    """
    Start gRPC server.

    Args:
        model_path: Path to model checkpoint
        port: Server port
        max_workers: Maximum concurrent workers
    """
    # Create server
    server = grpc.server(
        futures.ThreadPoolExecutor(max_workers=max_workers),
        options=[
            ('grpc.max_send_message_length', 50 * 1024 * 1024),  # 50MB
            ('grpc.max_receive_message_length', 50 * 1024 * 1024),
            ('grpc.so_reuseport', 1),
            ('grpc.use_local_subchannel_pool', 1)
        ]
    )

    # Initialize model manager
    model_manager = ModelManager(model_path=model_path)

    # Add servicer to server
    prediction_pb2_grpc.add_PredictionServiceServicer_to_server(
        PredictionServicer(model_manager), server
    )

    # Start server
    server.add_insecure_port(f'[::]:{port}')
    server.start()

    logger.info(f"gRPC server started on port {port}")
    logger.info(f"Max workers: {max_workers}")

    try:
        server.wait_for_termination()
    except KeyboardInterrupt:
        logger.info("Shutting down gRPC server...")
        server.stop(0)

import grpc
import prediction_pb2
import prediction_pb2_grpc
import numpy as np
import time

class PredictionClient:
    """
    gRPC client for predictions.

    Features:
    - Connection pooling
    - Automatic retry
    - Load balancing
    - Timeout handling
    """

    def __init__(self, host: str = 'localhost', port: int = 50051):
        self.address = f'{host}:{port}'

        # Create channel with keepalive
        self.channel = grpc.insecure_channel(
            self.address,
            options=[
                ('grpc.keepalive_time_ms', 10000),
                ('grpc.keepalive_timeout_ms', 5000),
                ('grpc.keepalive_permit_without_calls', True),
                ('grpc.http2.max_pings_without_data', 0),
            ]
        )

        self.stub = prediction_pb2_grpc.PredictionServiceStub(self.channel)

    def predict(self, features: np.ndarray, model_version: str = "v1", timeout: float = 5.0):
        """
        Make single prediction.

        Args:
            features: Input features
            model_version: Model version to use
            timeout: Request timeout in seconds

        Returns:
            Prediction response
        """
        request = prediction_pb2.PredictRequest(
            features=features.tolist(),
            model_version=model_version
        )

        try:
            response = self.stub.Predict(request, timeout=timeout)
            return {
                'prediction': response.prediction,
                'probabilities': list(response.probabilities),
                'model_version': response.model_version,
                'latency_ms': response.latency_ms
            }

        except grpc.RpcError as e:
            print(f"RPC error: {e.code()}: {e.details()}")
            raise

    def predict_batch(self, batch: np.ndarray, model_version: str = "v1", timeout: float = 10.0):
        """Make batch prediction."""
        # Build batch request
        features_list = [
            prediction_pb2.Features(values=row.tolist())
            for row in batch
        ]

        request = prediction_pb2.BatchPredictRequest(
            batch=features_list,
            model_version=model_version

\begin{lstlisting}[style=python, caption=Dynamic batch serving for throughput optimization]
import torch
import asyncio
from dataclasses import dataclass
from typing import List, Any
import time

@dataclass
class PendingRequest:
    """Request waiting in batch queue."""
    features: torch.Tensor
    future: asyncio.Future
    arrival_time: float

class DynamicBatchServer:
    """
    Dynamic batching for high-throughput serving.

    Strategy:
    - Accumulate requests up to max_batch_size
    - Process when batch full or timeout reached
    - Balance latency vs throughput

    Benefits:
    - 10-50x higher throughput than online serving
    - Better GPU utilization
    - Amortized inference cost
    """

    def __init__(self,
                 model: nn.Module,
                 max_batch_size: int = 32,
                 max_wait_ms: float = 10.0,
                 device: str = "cuda"):
        self.model = model.to(device).eval()
        self.device = device
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms / 1000  # Convert to seconds

        self.pending_requests: List[PendingRequest] = []
        self.lock = asyncio.Lock()
        self.batch_event = asyncio.Event()

        # Start batch processor
        self.processing_task = None

    async def start(self):
        """Start batch processing loop."""
        self.processing_task = asyncio.create_task(self._process_batches())

    async def stop(self):
        """Stop batch processing."""
        if self.processing_task:
            self.processing_task.cancel()

    async def predict(self, features: torch.Tensor) -> Dict:
        """
        Add request to batch queue and wait for result.

        Args:
            features: Input tensor

        Returns:
            Prediction result
        """
        # Create future for this request
        future = asyncio.Future()

        request = PendingRequest(
            features=features,
            future=future,
            arrival_time=time.time()
        )

        # Add to queue
        async with self.lock:
            self.pending_requests.append(request)

            # Trigger batch processing if queue full
            if len(self.pending_requests) >= self.max_batch_size:
                self.batch_event.set()

        # Wait for result
        return await future

    async def _process_batches(self):
        """Background task to process batches."""
        while True:
            try:
                # Wait for batch to be ready
                await asyncio.wait_for(
                    self.batch_event.wait(),
                    timeout=self.max_wait_ms
                )
            except asyncio.TimeoutError:
                # Timeout reached - process whatever we have
                pass

            # Get batch
            async with self.lock:
                if len(self.pending_requests) == 0:
                    self.batch_event.clear()
                    continue

                # Take up to max_batch_size requests
                batch_requests = self.pending_requests[:self.max_batch_size]
                self.pending_requests = self.pending_requests[self.max_batch_size:]

                # Clear event if queue empty
                if len(self.pending_requests) < self.max_batch_size:
                    self.batch_event.clear()

            # Process batch
            await self._execute_batch(batch_requests)

    async def _execute_batch(self, requests: List[PendingRequest]):
        """Execute inference on batch."""
        if len(requests) == 0:
            return

        start_time = time.time()

        try:
            # Stack features into batch
            features_batch = torch.stack([r.features for r in requests]).to(self.device)

            # Run batch inference
            with torch.no_grad():
                outputs = self.model(features_batch)
                probabilities = torch.softmax(outputs, dim=1)
                predictions = torch.argmax(outputs, dim=1)

            # Synchronize
            if self.device == "cuda":
                torch.cuda.synchronize()

            latency_ms = (time.time() - start_time) * 1000

            # Return results to futures
            for idx, request in enumerate(requests):
                result = {
                    'prediction': predictions[idx].item(),
                    'probabilities': probabilities[idx].cpu().numpy().tolist(),
                    'latency_ms': latency_ms,
                    'batch_size': len(requests),
                    'wait_time_ms': (start_time - request.arrival_time) * 1000
                }
                request.future.set_result(result)

        except Exception as e:
            # Set exception on all futures
            for request in requests:
                request.future.set_exception(e)

        output = self.model(features)
        prediction = output.argmax(dim=1).item()
        probabilities = torch.softmax(output, dim=1).cpu().numpy().tolist()[0]

        return {
            'prediction': prediction,
            'probabilities': probabilities,
            'timestamp': time.time()
        }

        Optimize ONNX model.

        Optimizations:
        - Constant folding
        - Operator fusion
        - Redundant node elimination

        model = onnx.load(input_path)

            input_path,
            output_path,
            weight_type=QuantType.QInt8

class ONNXRuntimePredictor:
    """
    High-performance inference with ONNX Runtime.

    ONNX Runtime provides:
    - Optimized kernels for CPU/GPU
    - Automatic batching
    - Multi-threading
    """

    def __init__(self, model_path: str, device: str = "cuda"):
        self.device = device

        # Create session
        providers = ['CUDAExecutionProvider', 'CPUExecutionProvider'] if device == "cuda" else ['CPUExecutionProvider']

        self.session = ort.InferenceSession(
            model_path,
            providers=providers
        )

        # Get input/output names
        self.input_name = self.session.get_inputs()[0].name
        self.output_name = self.session.get_outputs()[0].name

        print(f"ONNX Runtime session created")
        print(f"Device: {self.session.get_providers()}")

    def predict(self, features: np.ndarray) -> Dict:
        """
        Run inference with ONNX Runtime.

        Args:
            features: Input features (numpy array)

        Returns:
            Prediction results
        """
        start_time = time.time()

        # Ensure correct shape
        if features.ndim == 1:
            features = features.reshape(1, -1)

        features = features.astype(np.float32)

        # Run inference
        outputs = self.session.run(
            [self.output_name],
            {self.input_name: features}
        )[0]

        # Post-process
        probabilities = self._softmax(outputs)
        prediction = probabilities.argmax(axis=1)[0]

        latency_ms = (time.time() - start_time) * 1000

        return {
            'prediction': int(prediction),
            'probabilities': probabilities[0].tolist(),
            'latency_ms': latency_ms
        }

    @staticmethod
    def _softmax(x: np.ndarray) -> np.ndarray:
        """Compute softmax."""
        exp_x = np.exp(x - np.max(x, axis=1, keepdims=True))
        return exp_x / exp_x.sum(axis=1, keepdims=True)

import tensorrt as trt
import pycuda.driver as cuda
import pycuda.autoinit
import numpy as np

class TensorRTOptimizer:
    """
    Optimize models with TensorRT for NVIDIA GPUs.

    Benefits:
    - 5-10x faster inference on NVIDIA GPUs
    - Automatic kernel fusion
    - Mixed precision (FP16/INT8)
    - Layer-level optimizations
    """

    def __init__(self):
        self.logger = trt.Logger(trt.Logger.WARNING)
        self.builder = trt.Builder(self.logger)
        self.network = None
        self.config = None
        self.engine = None

    def build_engine_from_onnx(self,
                               onnx_path: str,
                               fp16_mode: bool = True,
                               int8_mode: bool = False,
                               max_batch_size: int = 32) -> trt.ICudaEngine:
        """
        Build TensorRT engine from ONNX model.

        Args:
            onnx_path: Path to ONNX model
            fp16_mode: Enable FP16 precision
            int8_mode: Enable INT8 precision (requires calibration)
            max_batch_size: Maximum batch size

        Returns:
            TensorRT engine
        """
        # Create network
        network_flags = 1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
        self.network = self.builder.create_network(network_flags)

        # Parse ONNX
        parser = trt.OnnxParser(self.network, self.logger)

        with open(onnx_path, 'rb') as f:
            if not parser.parse(f.read()):
                print("ERROR: Failed to parse ONNX model")
                for error in range(parser.num_errors):
                    print(parser.get_error(error))
                return None

        # Create builder config
        self.config = self.builder.create_builder_config()

        # Set max workspace size (4GB)
        self.config.max_workspace_size = 4 * (1 << 30)

        # Enable optimizations
        if fp16_mode and self.builder.platform_has_fast_fp16:
            self.config.set_flag(trt.BuilderFlag.FP16)
            print("FP16 mode enabled")

        if int8_mode and self.builder.platform_has_fast_int8:
            self.config.set_flag(trt.BuilderFlag.INT8)
            # Note: INT8 requires calibration dataset
            print("INT8 mode enabled (calibration required)")

        # Set optimization profile
        profile = self.builder.create_optimization_profile()

        # Set dynamic shape range
        input_name = self.network.get_input(0).name
        profile.set_shape(
            input_name,
            min=(1, 784),      # Min batch size
            opt=(8, 784),      # Optimal batch size
            max=(max_batch_size, 784)  # Max batch size
        )

        self.config.add_optimization_profile(profile)

        # Build engine
        print("Building TensorRT engine... This may take a few minutes.")
        self.engine = self.builder.build_engine(self.network, self.config)

        if self.engine is None:
            print("ERROR: Failed to build engine")
            return None

        print("TensorRT engine built successfully")
        return self.engine

    def save_engine(self, engine: trt.ICudaEngine, output_path: str):
        """Save TensorRT engine to file."""
        with open(output_path, 'wb') as f:
            f.write(engine.serialize())
        print(f"Engine saved to {output_path}")

    def load_engine(self, engine_path: str) -> trt.ICudaEngine:
        """Load TensorRT engine from file."""
        runtime = trt.Runtime(self.logger)

        with open(engine_path, 'rb') as f:
            engine = runtime.deserialize_cuda_engine(f.read())

        print(f"Engine loaded from {engine_path}")
        return engine

class TensorRTPredictor:
    """
    High-performance inference with TensorRT.

    Performance:
    - FP32: 2-3x faster than PyTorch
    - FP16: 5-7x faster than PyTorch
    - INT8: 8-10x faster than PyTorch
    """

    def __init__(self, engine: trt.ICudaEngine):
        self.engine = engine
        self.context = engine.create_execution_context()

        # Allocate buffers
        self.inputs = []
        self.outputs = []
        self.bindings = []
        self.stream = cuda.Stream()

        for binding in engine:
            size = trt.volume(engine.get_binding_shape(binding))
            dtype = trt.nptype(engine.get_binding_dtype(binding))

            # Allocate host and device buffers
            host_mem = cuda.pagelocked_empty(size, dtype)
            device_mem = cuda.mem_alloc(host_mem.nbytes)

            self.bindings.append(int(device_mem))

            if engine.binding_is_input(binding):
                self.inputs.append({'host': host_mem, 'device': device_mem})
            else:
                self.outputs.append({'host': host_mem, 'device': device_mem})

    def predict(self, features: np.ndarray) -> Dict:
        """
        Run inference with TensorRT.

        Args:
            features: Input features

        Returns:
            Prediction results
        """
        start_time = time.time()

        # Ensure correct shape
        if features.ndim == 1:
            features = features.reshape(1, -1)

        features = features.astype(np.float32)

        # Copy input to device
        np.copyto(self.inputs[0]['host'], features.ravel())
        cuda.memcpy_htod_async(
            self.inputs[0]['device'],
            self.inputs[0]['host'],
            self.stream
        )

        # Run inference
        self.context.execute_async_v2(
            bindings=self.bindings,
            stream_handle=self.stream.handle
        )

        # Copy output to host
        cuda.memcpy_dtoh_async(
            self.outputs[0]['host'],
            self.outputs[0]['device'],
            self.stream
        )

        # Synchronize
        self.stream.synchronize()

        # Get output
        output = self.outputs[0]['host'].reshape(features.shape[0], -1)

        # Post-process
        probabilities = self._softmax(output)
        prediction = probabilities.argmax(axis=1)[0]

        latency_ms = (time.time() - start_time) * 1000

        return {
            'prediction': int(prediction),
            'probabilities': probabilities[0].tolist(),
            'latency_ms': latency_ms
        }

    @staticmethod
    def _softmax(x: np.ndarray) -> np.ndarray:
        """Compute softmax."""
        exp_x = np.exp(x - np.max(x, axis=1, keepdims=True))
        return exp_x / exp_x.sum(axis=1, keepdims=True)

import time
from contextlib import contextmanager
from typing import Dict, List
import statistics

class LatencyProfiler:
    """
    Profile latency breakdown for serving pipeline.

    Tracks:
    - Preprocessing time
    - Model inference time
    - Postprocessing time
    - Total end-to-end latency
    """

    def __init__(self):
        self.measurements: Dict[str, List[float]] = {
            'preprocessing': [],
            'inference': [],
            'postprocessing': [],
            'total': []
        }

    @contextmanager
    def measure(self, stage: str):
        """Context manager to measure stage latency."""
        start = time.perf_counter()
        try:
            yield
        finally:
            elapsed = (time.perf_counter() - start) * 1000  # ms
            self.measurements[stage].append(elapsed)

    def get_stats(self) -> Dict:
        """Get latency statistics."""
        stats = {}

        for stage, measurements in self.measurements.items():
            if len(measurements) == 0:
                continue

            stats[stage] = {
                'mean': statistics.mean(measurements),
                'median': statistics.median(measurements),
                'p95': self._percentile(measurements, 95),
                'p99': self._percentile(measurements, 99),
                'min': min(measurements),
                'max': max(measurements),
                'count': len(measurements)
            }

        return stats

    @staticmethod
    def _percentile(data: List[float], percentile: float) -> float:
        """Calculate percentile."""
        sorted_data = sorted(data)
        index = int(len(sorted_data) * percentile / 100)
        return sorted_data[min(index, len(sorted_data) - 1)]

    def print_report(self):
        """Print profiling report."""
        stats = self.get_stats()

        print("\n" + "=" * 80)
        print("LATENCY PROFILING REPORT")
        print("=" * 80)

        for stage, metrics in stats.items():
            print(f"\n{stage.upper()}:")
            print(f"  Mean:     {metrics['mean']:.2f} ms")
            print(f"  Median:   {metrics['median']:.2f} ms")
            print(f"  P95:      {metrics['p95']:.2f} ms")
            print(f"  P99:      {metrics['p99']:.2f} ms")
            print(f"  Min:      {metrics['min']:.2f} ms")
            print(f"  Max:      {metrics['max']:.2f} ms")
            print(f"  Samples:  {metrics['count']}")

import torch
import torch.nn as nn
from typing import List
import numpy as np

class ThroughputOptimizer:
    """
    Optimize serving throughput.

    Techniques:
    - Optimal batch size selection
    - Multi-GPU serving
    - Model parallelism
    - Request queueing
    """

    @staticmethod
    def find_optimal_batch_size(model: nn.Module,
                                input_shape: tuple,
                                device: str = "cuda",
                                max_batch_size: int = 128) -> int:
        """
        Find optimal batch size for maximum throughput.

        Strategy:
        - Test increasing batch sizes
        - Measure throughput (samples/sec)
        - Return batch size with highest throughput
        """
        model = model.to(device).eval()
        results = []

        batch_sizes = [1, 2, 4, 8, 16, 32, 64, 128]
        batch_sizes = [bs for bs in batch_sizes if bs <= max_batch_size]

        print("Finding optimal batch size...")

        for batch_size in batch_sizes:
            try:
                # Create dummy batch
                dummy_input = torch.randn(batch_size, *input_shape).to(device)

                # Warmup
                with torch.no_grad():
                    for _ in range(10):
                        _ = model(dummy_input)

                # Measure throughput
                n_iterations = 100
                start_time = time.time()

                with torch.no_grad():
                    for _ in range(n_iterations):
                        _ = model(dummy_input)

                # Synchronize
                if device == "cuda":
                    torch.cuda.synchronize()

                elapsed = time.time() - start_time
                throughput = (n_iterations * batch_size) / elapsed

                results.append({
                    'batch_size': batch_size,
                    'throughput': throughput,
                    'latency': elapsed / n_iterations * 1000
                })

                print(f"  Batch size {batch_size:3d}: {throughput:8.1f} samples/sec, "
                      f"{elapsed / n_iterations * 1000:.2f} ms/batch")

            except RuntimeError as e:
                if "out of memory" in str(e):
                    print(f"  Batch size {batch_size:3d}: OOM")
                    break
                raise

        # Find optimal
        if len(results) == 0:
            return 1

        optimal = max(results, key=lambda x: x['throughput'])
        print(f"\nOptimal batch size: {optimal['batch_size']}")
        print(f"Max throughput: {optimal['throughput']:.1f} samples/sec")

        return optimal['batch_size']

class MultiGPUServing:
    """
    Serve model across multiple GPUs for higher throughput.

    Strategy:
    - Replicate model on each GPU
    - Round-robin request distribution
    - Independent inference on each GPU
    """

    def __init__(self, model: nn.Module, n_gpus: int = None):
        if n_gpus is None:
            n_gpus = torch.cuda.device_count()

        self.n_gpus = n_gpus
        self.models = []

        # Create model replica on each GPU
        for gpu_id in range(n_gpus):
            model_replica = model.to(f'cuda:{gpu_id}')
            model_replica.eval()
            self.models.append(model_replica)

        self.current_gpu = 0
        print(f"MultiGPU serving initialized with {n_gpus} GPUs")

    @torch.no_grad()
    def predict(self, features: np.ndarray) -> Dict:
        """
        Predict using next available GPU (round-robin).

        Args:
            features: Input features

        Returns:
            Prediction result
        """
        # Select GPU
        gpu_id = self.current_gpu
        self.current_gpu = (self.current_gpu + 1) % self.n_gpus

        # Move to GPU and run inference
        device = f'cuda:{gpu_id}'
        x = torch.from_numpy(features).float().unsqueeze(0).to(device)

        output = self.models[gpu_id](x)
        prediction = output.argmax(dim=1).item()
        probabilities = torch.softmax(output, dim=1).cpu().numpy().tolist()[0]

        return {
            'prediction': prediction,
            'probabilities': probabilities,
            'gpu_id': gpu_id
        }

import torch
import gc

class MemoryOptimizer:
    """
    Optimize memory usage in serving.

    Techniques:
    - Model quantization
    - Gradient disabled
    - Memory pooling
    - Periodic garbage collection
    """

    @staticmethod
    def optimize_model_memory(model: nn.Module) -> nn.Module:
        """Optimize model for minimal memory footprint."""

        # Ensure eval mode (no gradients)
        model.eval()

        # Disable gradient computation globally
        for param in model.parameters():
            param.requires_grad = False

        # Use inference mode for better performance
        model = torch.jit.optimize_for_inference(
            torch.jit.script(model)
        )

        return model

    @staticmethod
    def monitor_memory():
        """Monitor GPU memory usage."""
        if torch.cuda.is_available():
            allocated = torch.cuda.memory_allocated() / 1024**3  # GB
            reserved = torch.cuda.memory_reserved() / 1024**3
            max_allocated = torch.cuda.max_memory_allocated() / 1024**3

            print(f"GPU Memory:")
            print(f"  Allocated: {allocated:.2f} GB")
            print(f"  Reserved:  {reserved:.2f} GB")
            print(f"  Peak:      {max_allocated:.2f} GB")

    @staticmethod
    def clear_memory():
        """Clear unused memory."""
        gc.collect()
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
            torch.cuda.synchronize()

    @staticmethod
    def setup_memory_pooling():
        """Configure memory allocator for better reuse."""
        if torch.cuda.is_available():
            # Use caching allocator
            torch.cuda.set_per_process_memory_fraction(0.9, device=0)

            # Pre-allocate memory pool
            torch.cuda.empty_cache()

\textbf{Serving Frameworks:}

\begin{itemize}

    \item Kubernetes enables horizontal autoscaling based on CPU, memory, and custom metrics

\textbf{Performance Optimization:}
\begin{itemize}

    \item Optimal batch size selection maximizes throughput for given hardware
    \item Multi-GPU serving increases capacity with model replication
    \item Memory optimization reduces footprint through quantization and efficient allocation
\end{itemize}
