\section{Introduction}

MLOps platforms provide integrated environments for developing, deploying, and managing machine learning models at scale. The platform landscape ranges from open-source frameworks to fully managed cloud services, each with distinct capabilities, trade-offs, and use cases.

\textbf{MLOps Platform Categories:}

\begin{itemize}
    \item \textbf{Open-Source Frameworks:} Kubeflow, MLflow, Metaflow (flexibility, customization)
    \item \textbf{Cloud-Native Platforms:} AWS SageMaker, Google Vertex AI, Azure ML (managed services)
    \item \textbf{Specialized Tools:} DataRobot, Dataiku, Databricks (specific workflows)
    \item \textbf{Custom Solutions:} In-house platforms built on orchestrators (full control)
\end{itemize}

\textbf{Platform Selection Factors:}

\begin{enumerate}
    \item \textbf{Cloud Strategy:} Single-cloud, multi-cloud, or hybrid deployments
    \item \textbf{Team Size:} Platform operations overhead vs team capacity
    \item \textbf{Use Cases:} Batch inference, real-time serving, edge deployment
    \item \textbf{Existing Infrastructure:} Kubernetes, cloud services, on-premises
    \item \textbf{Compliance:} Data residency, security certifications, audit requirements
    \item \textbf{Budget:} TCO including licensing, infrastructure, and operations
    \item \textbf{Vendor Lock-in:} Portability and exit strategy considerations
\end{enumerate}

# pipeline.py
"""
End-to-end ML pipeline with Kubeflow Pipelines SDK
Covers data processing, training, evaluation, and deployment
"""

import kfp
from kfp import dsl
from kfp.components import create_component_from_func, InputPath, OutputPath
from typing import NamedTuple

# Component 1: Data Ingestion
@create_component_from_func
def ingest_data(
    data_source: str,
    output_dataset: OutputPath('Dataset')
) -> NamedTuple('Outputs', [('num_samples', int), ('num_features', int)]):
    """Ingest data from source and save as dataset"""

    import pandas as pd
    import os

    # Load data
    if data_source.startswith('s3://'):
        import boto3
        # Load from S3
        df = pd.read_csv(data_source)
    else:
        df = pd.read_csv(data_source)

    # Save dataset
    os.makedirs(os.path.dirname(output_dataset), exist_ok=True)
    df.to_csv(output_dataset, index=False)

    return (len(df), len(df.columns))

# Component 2: Data Validation
@create_component_from_func
def validate_data(
    dataset: InputPath('Dataset'),
    validation_report: OutputPath('ValidationReport')
) -> NamedTuple('Outputs', [('is_valid', bool), ('quality_score', float)]):
    """Validate data quality and schema"""

    import pandas as pd
    import json
    import numpy as np

    df = pd.read_csv(dataset)

    # Validation checks
    checks = {
        'null_percentage': df.isnull().sum().sum() / (len(df) * len(df.columns)),
        'duplicate_rows': len(df[df.duplicated()]),
        'outliers_percentage': 0.0  # Simplified
    }

    # Calculate quality score
    quality_score = 1.0 - checks['null_percentage']
    is_valid = quality_score > 0.95

    # Save validation report
    with open(validation_report, 'w') as f:
        json.dump(checks, f, indent=2)

    return (is_valid, float(quality_score))

# Component 3: Feature Engineering
@create_component_from_func
def engineer_features(
    dataset: InputPath('Dataset'),
    output_features: OutputPath('Features'),
    target_column: str = 'target'
):
    """Engineer features from raw data"""

    import pandas as pd
    from sklearn.preprocessing import StandardScaler
    import pickle

    df = pd.read_csv(dataset)

    # Separate features and target
    X = df.drop(columns=[target_column])
    y = df[target_column]

    # Feature engineering
    # 1. Handle missing values
    X = X.fillna(X.median())

    # 2. Scale features
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    # 3. Combine back
    feature_df = pd.DataFrame(X_scaled, columns=X.columns)
    feature_df[target_column] = y.values

    # Save engineered features
    feature_df.to_csv(output_features, index=False)

    print(f"Engineered {len(feature_df.columns)-1} features")

# Component 4: Train-Test Split
@create_component_from_func
def split_data(
    features: InputPath('Features'),
    train_data: OutputPath('TrainData'),
    test_data: OutputPath('TestData'),
    test_size: float = 0.2,
    random_state: int = 42
):
    """Split data into train and test sets"""

    import pandas as pd
    from sklearn.model_selection import train_test_split

    df = pd.read_csv(features)

    train_df, test_df = train_test_split(
        df,
        test_size=test_size,
        random_state=random_state
    )

    train_df.to_csv(train_data, index=False)
    test_df.to_csv(test_data, index=False)

    print(f"Train samples: {len(train_df)}, Test samples: {len(test_df)}")

# Component 5: Model Training
@create_component_from_func
def train_model(
    train_data: InputPath('TrainData'),
    model_output: OutputPath('Model'),
    model_type: str = 'random_forest',
    n_estimators: int = 100,
    max_depth: int = 10
) -> NamedTuple('Outputs', [('train_accuracy', float), ('train_time', float)]):
    """Train ML model"""

    import pandas as pd
    import pickle
    import time
    from sklearn.ensemble import RandomForestClassifier
    from sklearn.linear_model import LogisticRegression
    from sklearn.metrics import accuracy_score

    # Load data
    df = pd.read_csv(train_data)
    X_train = df.drop(columns=['target'])
    y_train = df['target']

    # Select and train model
    start_time = time.time()

    if model_type == 'random_forest':
        model = RandomForestClassifier(
            n_estimators=n_estimators,
            max_depth=max_depth,
            random_state=42
        )
    else:
        model = LogisticRegression(max_iter=1000, random_state=42)

    model.fit(X_train, y_train)
    train_time = time.time() - start_time

    # Calculate training accuracy
    train_pred = model.predict(X_train)
    train_acc = accuracy_score(y_train, train_pred)

    # Save model
    with open(model_output, 'wb') as f:
        pickle.dump(model, f)

    print(f"Model trained: {model_type}, Accuracy: {train_acc:.4f}")

    return (float(train_acc), float(train_time))

# Component 6: Model Evaluation
@create_component_from_func
def evaluate_model(
    model: InputPath('Model'),
    test_data: InputPath('TestData'),
    metrics_output: OutputPath('Metrics')
) -> NamedTuple('Outputs', [
    ('test_accuracy', float),
    ('test_f1', float),
    ('test_auc', float)
]):
    """Evaluate model on test data"""

    import pandas as pd
    import pickle
    import json
    from sklearn.metrics import (
        accuracy_score,
        f1_score,
        roc_auc_score,
        classification_report,
        confusion_matrix
    )

    # Load model and data
    with open(model, 'rb') as f:
        model = pickle.load(f)

    df = pd.read_csv(test_data)
    X_test = df.drop(columns=['target'])
    y_test = df['target']

    # Make predictions
    y_pred = model.predict(X_test)
    y_proba = model.predict_proba(X_test)[:, 1] if hasattr(model, 'predict_proba') else y_pred

    # Calculate metrics
    test_acc = accuracy_score(y_test, y_pred)
    test_f1 = f1_score(y_test, y_pred, average='weighted')
    test_auc = roc_auc_score(y_test, y_proba)

    # Detailed metrics
    metrics = {
        'accuracy': float(test_acc),
        'f1_score': float(test_f1),
        'auc_roc': float(test_auc),
        'classification_report': classification_report(y_test, y_pred, output_dict=True),
        'confusion_matrix': confusion_matrix(y_test, y_pred).tolist()
    }

    # Save metrics
    with open(metrics_output, 'w') as f:
        json.dump(metrics, f, indent=2)

    print(f"Test Accuracy: {test_acc:.4f}, F1: {test_f1:.4f}, AUC: {test_auc:.4f}")

    return (float(test_acc), float(test_f1), float(test_auc))

# Component 7: Model Deployment Decision
@create_component_from_func
def deployment_decision(
    test_accuracy: float,
    test_f1: float,
    accuracy_threshold: float = 0.85,
    f1_threshold: float = 0.80
) -> NamedTuple('Outputs', [('should_deploy', bool), ('decision_reason', str)]):
    """Decide whether to deploy model based on metrics"""

    if test_accuracy >= accuracy_threshold and test_f1 >= f1_threshold:
        return (True, f"Metrics above threshold: acc={test_accuracy:.4f}, f1={test_f1:.4f}")
    else:
        return (False, f"Metrics below threshold: acc={test_accuracy:.4f}, f1={test_f1:.4f}")

# Component 8: Model Registration
@create_component_from_func
def register_model(
    model: InputPath('Model'),
    model_name: str,
    model_version: str,
    metrics: InputPath('Metrics')
):
    """Register model in model registry"""

    import mlflow
    import pickle
    import json

    # Load model and metrics
    with open(model, 'rb') as f:
        model_obj = pickle.load(f)

    with open(metrics, 'r') as f:
        metrics_dict = json.load(f)

    # Register with MLflow
    mlflow.set_tracking_uri("http://mlflow-server:5000")

    with mlflow.start_run():
        # Log metrics
        mlflow.log_metrics({
            'test_accuracy': metrics_dict['accuracy'],
            'test_f1': metrics_dict['f1_score'],
            'test_auc': metrics_dict['auc_roc']
        })

        # Log model
        mlflow.sklearn.log_model(
            model_obj,
            "model",
            registered_model_name=model_name
        )

        # Add version tag
        mlflow.set_tag("version", model_version)

    print(f"Model registered: {model_name} version {model_version}")

# Define the Pipeline
@dsl.pipeline(
    name='ML Training Pipeline',
    description='End-to-end ML pipeline with data processing, training, and deployment'
)
def ml_training_pipeline(
    data_source: str = 's3://ml-data/dataset.csv',
    model_type: str = 'random_forest',
    n_estimators: int = 100,
    max_depth: int = 10,
    test_size: float = 0.2,
    accuracy_threshold: float = 0.85,
    model_name: str = 'classifier-model',
    model_version: str = 'v1.0'
):
    """Complete ML training pipeline"""

    # Step 1: Ingest data
    ingest_task = ingest_data(data_source=data_source)

    # Step 2: Validate data
    validate_task = validate_data(dataset=ingest_task.outputs['output_dataset'])

    # Step 3: Engineer features (only if validation passes)
    with dsl.Condition(validate_task.outputs['is_valid'] == True):
        engineer_task = engineer_features(
            dataset=ingest_task.outputs['output_dataset']
        )

        # Step 4: Split data
        split_task = split_data(
            features=engineer_task.outputs['output_features'],
            test_size=test_size
        )

        # Step 5: Train model
        train_task = train_model(
            train_data=split_task.outputs['train_data'],
            model_type=model_type,
            n_estimators=n_estimators,
            max_depth=max_depth
        )

        # Step 6: Evaluate model
        eval_task = evaluate_model(
            model=train_task.outputs['model_output'],
            test_data=split_task.outputs['test_data']
        )

        # Step 7: Deployment decision
        decision_task = deployment_decision(
            test_accuracy=eval_task.outputs['test_accuracy'],
            test_f1=eval_task.outputs['test_f1'],
            accuracy_threshold=accuracy_threshold
        )

        # Step 8: Register model (only if deployment approved)
        with dsl.Condition(decision_task.outputs['should_deploy'] == True):
            register_task = register_model(
                model=train_task.outputs['model_output'],
                model_name=model_name,
                model_version=model_version,
                metrics=eval_task.outputs['metrics_output']
            )

# Compile and run the pipeline
if __name__ == '__main__':
    import kfp.compiler as compiler

    # Compile pipeline to YAML
    compiler.Compiler().compile(
        ml_training_pipeline,
        'ml_training_pipeline.yaml'
    )

    # Create client and run
    client = kfp.Client(host='http://localhost:8080')

    # Submit pipeline run
    run = client.create_run_from_pipeline_func(
        ml_training_pipeline,
        arguments={
            'data_source': 's3://ml-data/classification_data.csv',
            'model_type': 'random_forest',
            'n_estimators': 100,
            'max_depth': 10,
            'accuracy_threshold': 0.85,
            'model_name': 'production-classifier',
            'model_version': 'v1.0'
        }
    )

    print(f"Pipeline run created: {run.run_id}")

# sagemaker_pipeline.py
"""
Complete ML pipeline using AWS SageMaker
"""

import boto3
import sagemaker
from sagemaker.sklearn import SKLearn
from sagemaker.processing import ScriptProcessor
from sagemaker.workflow.pipeline import Pipeline
from sagemaker.workflow.steps import ProcessingStep, TrainingStep
from sagemaker.workflow.parameters import ParameterInteger, ParameterString
from sagemaker.inputs import TrainingInput
from sagemaker.model import Model
from sagemaker.workflow.model_step import ModelStep

# Initialize SageMaker session
sess = sagemaker.Session()
role = sagemaker.get_execution_role()
bucket = sess.default_bucket()
region = boto3.Session().region_name

# Define pipeline parameters
processing_instance_type = ParameterString(
    name="ProcessingInstanceType",
    default_value="ml.m5.xlarge"
)

training_instance_type = ParameterString(
    name="TrainingInstanceType",
    default_value="ml.m5.2xlarge"
)

model_approval_status = ParameterString(
    name="ModelApprovalStatus",
    default_value="PendingManualApproval"
)

# Step 1: Data Processing
script_processor = ScriptProcessor(
    image_uri=f"683313688378.dkr.ecr.{region}.amazonaws.com/sagemaker-scikit-learn:0.23-1-cpu-py3",
    role=role,
    instance_type=processing_instance_type,
    instance_count=1,
    command=["python3"]
)

processing_step = ProcessingStep(
    name="DataProcessing",
    processor=script_processor,
    inputs=[
        sagemaker.processing.ProcessingInput(
            source=f"s3://{bucket}/input/data.csv",
            destination="/opt/ml/processing/input"
        )
    ],
    outputs=[
        sagemaker.processing.ProcessingOutput(
            source="/opt/ml/processing/train",
            destination=f"s3://{bucket}/processed/train"
        ),
        sagemaker.processing.ProcessingOutput(
            source="/opt/ml/processing/test",
            destination=f"s3://{bucket}/processed/test"
        )
    ],
    code="preprocessing.py"
)

# Step 2: Model Training
sklearn_estimator = SKLearn(
    entry_point="train.py",
    role=role,
    instance_type=training_instance_type,
    instance_count=1,
    framework_version="0.23-1",
    hyperparameters={
        "n_estimators": 100,
        "max_depth": 10
    }
)

training_step = TrainingStep(
    name="ModelTraining",
    estimator=sklearn_estimator,
    inputs={
        "train": TrainingInput(
            s3_data=processing_step.properties.ProcessingOutputConfig.Outputs[
                "train"
            ].S3Output.S3Uri,
            content_type="text/csv"
        ),
        "test": TrainingInput(
            s3_data=processing_step.properties.ProcessingOutputConfig.Outputs[
                "test"
            ].S3Output.S3Uri,
            content_type="text/csv"
        )
    }
)

# Step 3: Model Evaluation
evaluation_processor = ScriptProcessor(
    image_uri=f"683313688378.dkr.ecr.{region}.amazonaws.com/sagemaker-scikit-learn:0.23-1-cpu-py3",
    role=role,
    instance_type=processing_instance_type,
    instance_count=1,
    command=["python3"]
)

evaluation_step = ProcessingStep(
    name="ModelEvaluation",
    processor=evaluation_processor,
    inputs=[
        sagemaker.processing.ProcessingInput(
            source=training_step.properties.ModelArtifacts.S3ModelArtifacts,
            destination="/opt/ml/processing/model"
        ),
        sagemaker.processing.ProcessingInput(
            source=processing_step.properties.ProcessingOutputConfig.Outputs[
                "test"
            ].S3Output.S3Uri,
            destination="/opt/ml/processing/test"
        )
    ],
    outputs=[
        sagemaker.processing.ProcessingOutput(
            source="/opt/ml/processing/evaluation",
            destination=f"s3://{bucket}/evaluation"
        )
    ],
    code="evaluate.py"
)

# Step 4: Model Registration
model = Model(
    image_uri=training_step.properties.AlgorithmSpecification.TrainingImage,
    model_data=training_step.properties.ModelArtifacts.S3ModelArtifacts,
    sagemaker_session=sess,
    role=role
)

model_step = ModelStep(
    name="RegisterModel",
    step_args=model.register(
        content_types=["text/csv"],
        response_types=["text/csv"],
        inference_instances=["ml.m5.xlarge"],
        transform_instances=["ml.m5.xlarge"],
        model_package_group_name="ml-model-package-group",
        approval_status=model_approval_status
    )
)

# Create Pipeline
pipeline = Pipeline(
    name="MLTrainingPipeline",
    parameters=[
        processing_instance_type,
        training_instance_type,
        model_approval_status
    ],
    steps=[
        processing_step,
        training_step,
        evaluation_step,
        model_step
    ]
)

# Create/Update Pipeline
pipeline.upsert(role_arn=role)

# Start Pipeline Execution
execution = pipeline.start()
execution.wait()

print(f"Pipeline execution completed: {execution.arn}")
\end{lstlisting}

@component(
    base_image="python:3.9",
    packages_to_install=["pandas", "scikit-learn", "google-cloud-storage"]
)
def preprocess_data(
    input_data_uri: str,
    train_data: Output[Dataset],
    test_data: Output[Dataset]
):
    import pandas as pd
    from sklearn.model_selection import train_test_split

    # Load data from GCS
    df = pd.read_csv(input_data_uri)

    # Preprocess
    df = df.dropna()

    # Split
    train_df, test_df = train_test_split(df, test_size=0.2, random_state=42)

    # Save
    train_df.to_csv(train_data.path, index=False)
    test_df.to_csv(test_data.path, index=False)

# Component 2: Training
@component(
    base_image="python:3.9",
    packages_to_install=["pandas", "scikit-learn", "joblib"]
)
def train_model(
    train_data: Input[Dataset],
    model: Output[Model],
    n_estimators: int = 100,
    max_depth: int = 10
):
    import pandas as pd
    from sklearn.ensemble import RandomForestClassifier
    import joblib

    # Load training data
    df = pd.read_csv(train_data.path)
    X_train = df.drop(columns=['target'])
    y_train = df['target']

    # Train model
    clf = RandomForestClassifier(
        n_estimators=n_estimators,
        max_depth=max_depth,
        random_state=42
    )
    clf.fit(X_train, y_train)

    # Save model
    joblib.dump(clf, model.path)

# Component 3: Evaluation
@component(
    base_image="python:3.9",
    packages_to_install=["pandas", "scikit-learn", "joblib"]
)
def evaluate_model(
    model: Input[Model],
    test_data: Input[Dataset],
    metrics: Output[Metrics]
):
    import pandas as pd
    from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
    import joblib

    # Load model and test data
    clf = joblib.load(model.path)
    df = pd.read_csv(test_data.path)
    X_test = df.drop(columns=['target'])
    y_test = df['target']

    # Evaluate
    y_pred = clf.predict(X_test)
    y_proba = clf.predict_proba(X_test)[:, 1]

    accuracy = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred, average='weighted')
    auc = roc_auc_score(y_test, y_proba)

    # Log metrics
    metrics.log_metric("accuracy", accuracy)
    metrics.log_metric("f1_score", f1)
    metrics.log_metric("auc", auc)

#!/usr/bin/env python3
"""
MLOps Platform Selection Framework
Systematic evaluation of platform options based on weighted criteria
"""

from dataclasses import dataclass
from typing import List, Dict
from enum import Enum

class PlatformType(Enum):
    KUBEFLOW = "Kubeflow"
    SAGEMAKER = "AWS SageMaker"
    VERTEX_AI = "Google Vertex AI"
    AZURE_ML = "Azure ML"
    CUSTOM = "Custom Platform"

@dataclass
class EvaluationCriterion:
    """Single evaluation criterion with weight"""
    name: str
    weight: float  # 0.0 to 1.0
    description: str

@dataclass
class PlatformScore:
    """Score for a platform on specific criterion"""
    criterion: str
    score: float  # 0 to 10
    justification: str

class PlatformEvaluator:
    """Evaluate and compare MLOps platforms"""

    def __init__(self):
        self.criteria = self._define_criteria()
        self.platforms = list(PlatformType)

    def _define_criteria(self) -> List[EvaluationCriterion]:
        """Define evaluation criteria with weights"""
        return [
            EvaluationCriterion(
                name="ease_of_use",
                weight=0.15,
                description="Learning curve and operational complexity"
            ),
            EvaluationCriterion(
                name="cost",
                weight=0.20,
                description="Total cost of ownership (infrastructure + operations)"
            ),
            EvaluationCriterion(
                name="scalability",
                weight=0.15,
                description="Ability to scale compute and handle large workloads"
            ),
            EvaluationCriterion(
                name="flexibility",
                weight=0.10,
                description="Customization and extensibility options"
            ),
            EvaluationCriterion(
                name="cloud_strategy",
                weight=0.15,
                description="Alignment with multi-cloud vs single-cloud strategy"
            ),
            EvaluationCriterion(
                name="feature_completeness",
                weight=0.10,
                description="Built-in capabilities (feature store, monitoring, etc.)"
            ),
            EvaluationCriterion(
                name="integration",
                weight=0.05,
                description="Integration with existing tools and infrastructure"
            ),
            EvaluationCriterion(
                name="compliance",
                weight=0.05,
                description="Security certifications and data governance"
            ),
            EvaluationCriterion(
                name="vendor_lock_in",
                weight=0.05,
                description="Exit strategy and portability (higher score = less lock-in)"
            )
        ]

    def evaluate_platform(
        self,
        platform: PlatformType,
        scores: Dict[str, float],
        justifications: Dict[str, str]
    ) -> float:
        """Calculate weighted score for a platform"""

        total_score = 0.0
        total_weight = sum(c.weight for c in self.criteria)

        print(f"\nEvaluating {platform.value}:")
        print("=" * 60)

        for criterion in self.criteria:
            if criterion.name not in scores:
                print(f"Warning: Missing score for {criterion.name}")
                continue

            score = scores[criterion.name]
            weighted_score = score * criterion.weight
            total_score += weighted_score

            justification = justifications.get(criterion.name, "No justification provided")

            print(f"\n{criterion.name.replace('_', ' ').title()}: {score}/10")
            print(f"  Weight: {criterion.weight:.2f}")
            print(f"  Weighted Score: {weighted_score:.2f}")
            print(f"  Justification: {justification}")

        # Normalize to 0-10 scale
        normalized_score = (total_score / total_weight) * 10 if total_weight > 0 else 0

        print(f"\n{'='*60}")
        print(f"Total Weighted Score: {normalized_score:.2f}/10")

        return normalized_score

                dependencies=["switch_production_traffic"],

                risk_level="low",
                mitigation="Staged shutdown with verification steps"
            )

    def generate_timeline(self) -> dict:
        """Generate project timeline"""

        timeline = {}
        for phase in MigrationPhase:
            phase_tasks = [t for t in self.tasks if t.phase == phase]
            total_days = sum(t.estimated_days for t in phase_tasks)
            timeline[phase.value] = {
                "tasks": len(phase_tasks),
                "estimated_days": total_days,
                "tasks_list": [t.name for t in phase_tasks]
            }

        return timeline

    def identify_critical_path(self) -> List[str]:
        """Identify tasks on critical path"""

        # Simplified: tasks with dependencies form critical path
        critical_tasks = []

        for task in self.tasks:
            if task.risk_level in ["high", "critical"]:
                critical_tasks.append(task.name)

        return critical_tasks

    def export_plan(self, filename: str):

        plan = {

#!/usr/bin/env python3
"""
Platform-agnostic ML pipeline abstraction
Enables easy migration between platforms
"""

from abc import ABC, abstractmethod
from typing import Dict, Any, List
from enum import Enum

class PlatformBackend(Enum):
    KUBEFLOW = "kubeflow"
    SAGEMAKER = "sagemaker"
    VERTEX_AI = "vertex_ai"
    AZURE_ML = "azure_ml"

class MLPipelineStep(ABC):
    """Abstract base class for pipeline steps"""

    def __init__(self, name: str, inputs: Dict[str, str], outputs: Dict[str, str]):
        self.name = name
        self.inputs = inputs
        self.outputs = outputs

    @abstractmethod
    def execute(self, **kwargs) -> Dict[str, Any]:
        """Execute the step"""
        pass

    @abstractmethod
    def to_kubeflow(self) -> dict:
        """Convert to Kubeflow component"""
        pass

    @abstractmethod
    def to_sagemaker(self) -> dict:
        """Convert to SageMaker step"""
        pass

    @abstractmethod
    def to_vertex_ai(self) -> dict:
        """Convert to Vertex AI component"""
        pass

class DataPreprocessingStep(MLPipelineStep):
    """Data preprocessing step with multi-platform support"""

    def __init__(self, script_path: str):
        super().__init__(
            name="data_preprocessing",
            inputs={"raw_data": "s3://bucket/raw"},
            outputs={"processed_data": "s3://bucket/processed"}
        )
        self.script_path = script_path

    def execute(self, **kwargs) -> Dict[str, Any]:
        """Platform-agnostic execution logic"""
        # Actual preprocessing code
        return {"status": "success", "output_path": self.outputs["processed_data"]}

    def to_kubeflow(self) -> dict:
        """Convert to Kubeflow component"""
        return {
            "name": self.name,
            "container": {
                "image": "ml-preprocessing:latest",
                "command": ["python", self.script_path],
                "args": [
                    "--input", self.inputs["raw_data"],
                    "--output", self.outputs["processed_data"]
                ]
            }
        }

    def to_sagemaker(self) -> dict:
        """Convert to SageMaker ProcessingStep"""
        return {
            "processor": {
                "image_uri": "ml-preprocessing:latest",
                "instance_type": "ml.m5.xlarge",
                "instance_count": 1
            },
            "inputs": [
                {
                    "source": self.inputs["raw_data"],
                    "destination": "/opt/ml/processing/input"
                }
            ],
            "outputs": [
                {
                    "source": "/opt/ml/processing/output",
                    "destination": self.outputs["processed_data"]
                }
            ],
            "code": self.script_path
        }

    def to_vertex_ai(self) -> dict:
        """Convert to Vertex AI component"""
        return {
            "name": self.name,
            "base_image": "python:3.9",
            "packages_to_install": ["pandas", "scikit-learn"],
            "inputs": {"raw_data": self.inputs["raw_data"]},
            "outputs": {"processed_data": self.outputs["processed_data"]}
        }

class MLPipeline:
    """Platform-agnostic ML pipeline"""

    def __init__(self, name: str, steps: List[MLPipelineStep]):
        self.name = name
        self.steps = steps

    def compile_for_platform(self, platform: PlatformBackend) -> dict:
        """Compile pipeline for specific platform"""

        if platform == PlatformBackend.KUBEFLOW:
            return self._compile_kubeflow()
        elif platform == PlatformBackend.SAGEMAKER:
            return self._compile_sagemaker()
        elif platform == PlatformBackend.VERTEX_AI:
            return self._compile_vertex_ai()
        else:
            raise ValueError(f"Unsupported platform: {platform}")

    def _compile_kubeflow(self) -> dict:
        """Compile to Kubeflow pipeline"""
        return {
            "apiVersion": "argoproj.io/v1alpha1",
            "kind": "Workflow",
            "metadata": {"name": self.name},
            "spec": {
                "entrypoint": "pipeline",
                "templates": [step.to_kubeflow() for step in self.steps]
            }
        }

    def _compile_sagemaker(self) -> dict:
        """Compile to SageMaker pipeline"""
        return {
            "PipelineName": self.name,
            "Steps": [step.to_sagemaker() for step in self.steps]
        }

    def _compile_vertex_ai(self) -> dict:
        """Compile to Vertex AI pipeline"""
        return {
            "name": self.name,
            "components": [step.to_vertex_ai() for step in self.steps]
        }

#!/usr/bin/env python3
"""
MLOps Platform Cost Management
Track, analyze, and optimize platform costs
"""

import boto3
from datetime import datetime, timedelta
from typing import Dict, List
import pandas as pd

class PlatformCostManager:
    """Manage and optimize platform costs"""

    def __init__(self, platform_type: str = "sagemaker"):
        self.platform = platform_type
        self.ce_client = boto3.client('ce')  # Cost Explorer

    def get_cost_by_service(self, days: int = 30) -> pd.DataFrame:
        """Get cost breakdown by service"""

        end_date = datetime.now()
        start_date = end_date - timedelta(days=days)

        response = self.ce_client.get_cost_and_usage(
            TimePeriod={
                'Start': start_date.strftime('%Y-%m-%d'),
                'End': end_date.strftime('%Y-%m-%d')
            },
            Granularity='DAILY',
            Metrics=['UnblendedCost'],
            GroupBy=[
                {'Type': 'SERVICE', 'Key': 'SERVICE'},
                {'Type': 'TAG', 'Key': 'Team'}
            ],
            Filter={
                'Tags': {
                    'Key': 'ManagedBy',
                    'Values': ['mlops-platform']
                }
            }
        )

        # Process results
        cost_data = []
        for result in response['ResultsByTime']:
            date = result['TimePeriod']['Start']
            for group in result['Groups']:
                service = group['Keys'][0]
                team = group['Keys'][1] if len(group['Keys']) > 1 else 'untagged'
                cost = float(group['Metrics']['UnblendedCost']['Amount'])

                cost_data.append({
                    'date': date,
                    'service': service,
                    'team': team,
                    'cost': cost
                })

        return pd.DataFrame(cost_data)

    def identify_cost_anomalies(self, threshold: float = 1.5) -> List[Dict]:
        """Identify unusual cost spikes"""

        df = self.get_cost_by_service(days=30)

        # Calculate daily cost by team
        daily_costs = df.groupby(['date', 'team'])['cost'].sum().reset_index()

        anomalies = []
        for team in daily_costs['team'].unique():
            team_costs = daily_costs[daily_costs['team'] == team]['cost']
            mean_cost = team_costs.mean()
            std_cost = team_costs.std()

            for idx, row in daily_costs[daily_costs['team'] == team].iterrows():
                if row['cost'] > mean_cost + (threshold * std_cost):
                    anomalies.append({
                        'date': row['date'],
                        'team': team,
                        'cost': row['cost'],
                        'expected_cost': mean_cost,
                        'deviation': (row['cost'] - mean_cost) / mean_cost * 100
                    })

        return anomalies

    def recommend_optimizations(self) -> List[Dict]:
        """Generate cost optimization recommendations"""

        recommendations = []

        # Check for idle training instances
        sagemaker = boto3.client('sagemaker')

        # Get training jobs
        training_jobs = sagemaker.list_training_jobs(
            StatusEquals='InProgress',
            MaxResults=100
        )

        for job in training_jobs['TrainingJobSummaries']:
            job_details = sagemaker.describe_training_job(
                TrainingJobName=job['TrainingJobName']
            )

            # Check if using on-demand instances
            instance_type = job_details['ResourceConfig']['InstanceType']
            if not instance_type.startswith('ml.'):
                continue

            # Recommend spot instances for long-running jobs
            duration = datetime.now() - job['CreationTime'].replace(tzinfo=None)
            if duration.total_seconds() > 3600:  # > 1 hour
                recommendations.append({
                    'type': 'spot_instances',
                    'resource': job['TrainingJobName'],
                    'current_instance': instance_type,
                    'estimated_savings': '70%',
                    'description': 'Switch to spot instances for long training jobs'
                })

        # Check for over-provisioned endpoints
        endpoints = sagemaker.list_endpoints(MaxResults=100)

        for endpoint in endpoints['Endpoints']:
            endpoint_details = sagemaker.describe_endpoint(
                EndpointName=endpoint['EndpointName']
            )

            # Get metrics
            cloudwatch = boto3.client('cloudwatch')
            metrics = cloudwatch.get_metric_statistics(
                Namespace='AWS/SageMaker',
                MetricName='ModelLatency',
                Dimensions=[
                    {'Name': 'EndpointName', 'Value': endpoint['EndpointName']},
                    {'Name': 'VariantName', 'Value': 'AllTraffic'}
                ],
                StartTime=datetime.now() - timedelta(days=7),
                EndTime=datetime.now(),
                Period=86400,
                Statistics=['Average', 'Maximum']
            )

            # Check if consistently low latency (potential right-sizing opportunity)
            if metrics['Datapoints']:
                avg_latency = sum(d['Average'] for d in metrics['Datapoints']) / len(metrics['Datapoints'])
                if avg_latency < 50:  # < 50ms
                    recommendations.append({
                        'type': 'right_sizing',
                        'resource': endpoint['EndpointName'],
                        'current_config': endpoint_details['ProductionVariants'][0]['InstanceType'],
                        'estimated_savings': '30%',
                        'description': 'Consider smaller instance type for low-latency workload'
                    })

        return recommendations

    def generate_cost_report(self, output_file: str):
        """Generate comprehensive cost report"""

        df = self.get_cost_by_service(days=30)
        anomalies = self.identify_cost_anomalies()
        optimizations = self.recommend_optimizations()

        # Summary statistics

        daily_avg = df.groupby('date')['cost'].sum().mean()
