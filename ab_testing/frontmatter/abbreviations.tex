\section*{Statistical Test Abbreviations}

\begin{description}[leftmargin=!,labelwidth=\widthof{\bfseries Welch's t-test}]
\item[t-test] Student's t-test for comparing means
\item[Welch's t-test] t-test for unequal variances
\item[z-test] Normal approximation test for large samples
\item[$\chi^2$ test] Chi-squared test for categorical data
\item[F-test] Test for comparing variances or ANOVA
\item[Mann-Whitney U] Non-parametric test for comparing distributions
\item[Wilcoxon] Non-parametric paired test
\item[Kolmogorov-Smirnov] Non-parametric test for comparing distributions
\item[Fisher's exact] Exact test for $2 \times 2$ contingency tables
\item[Kruskal-Wallis] Non-parametric alternative to one-way ANOVA
\end{description}

\section*{Common Distribution Abbreviations}

\begin{description}[leftmargin=!,labelwidth=\widthof{\bfseries Hypergeometric}]
\item[Bernoulli] Binary outcome distribution (success/failure)
\item[Binomial] Sum of independent Bernoulli trials
\item[Poisson] Count distribution for rare events
\item[Normal/Gaussian] Bell-shaped continuous distribution
\item[Uniform] Equal probability distribution
\item[Exponential] Continuous distribution for time between events
\item[Beta] Continuous distribution on $[0, 1]$, conjugate prior for Bernoulli
\item[Gamma] Continuous distribution, conjugate prior for Poisson rate
\item[t-distribution] Student's t-distribution for small samples
\item[Chi-squared] Distribution of sum of squared normal variables
\item[F-distribution] Ratio of chi-squared variables
\item[Multinomial] Generalization of binomial to multiple categories
\item[Dirichlet] Multivariate generalization of beta distribution
\item[Negative Binomial] Distribution of trials until $r$ successes
\item[Geometric] Distribution of trials until first success
\item[Hypergeometric] Sampling without replacement
\end{description}

\section*{Notes on Usage}

\begin{itemize}
\item \textbf{Context Matters}: Some abbreviations have different meanings in different contexts (e.g., "SE" can mean Standard Error or Software Engineering).

\item \textbf{Capitalization}: While abbreviations are typically capitalized, their expansions may not be (e.g., "PDF" stands for "probability density function," not "Probability Density Function").

\item \textbf{Plural Forms}: When pluralizing abbreviations, add lowercase "s" without an apostrophe (e.g., "KPIs," "PDFs").

\item \textbf{Regional Variations}: Some terms have different names in different regions or disciplines (e.g., "conversion rate" vs. "CVR").

\item \textbf{Domain-Specific}: Many abbreviations are specific to online experimentation and may have different meanings in other fields.
\end{itemize}
