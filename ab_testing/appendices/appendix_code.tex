@dataclass
class ExperimentConfig:
    """Configuration for A/B test experiment."""

    experiment_id: str
    name: str
    description: str
    hypothesis: str

    # Design parameters
    variants: List[str]
    primary_metric: str
    secondary_metrics: List[str] = field(default_factory=list)
    guardrail_metrics: List[str] = field(default_factory=list)

    # Statistical parameters
    alpha: float = 0.05
    power: float = 0.80
    mde: Optional[float] = None  # Minimum detectable effect

    # Sample size
    sample_size_per_variant: Optional[int] = None

    # Duration
    start_date: Optional[datetime] = None
    end_date: Optional[datetime] = None

    # Traffic allocation
    traffic_allocation: Dict[str, float] = field(default_factory=dict)

    # Metadata
    owner: str = ""
    tags: List[str] = field(default_factory=list)

    def __post_init__(self):
        """Validate configuration."""
        if len(self.variants) < 2:
            raise ValueError("At least 2 variants required")

        if self.traffic_allocation:
            total = sum(self.traffic_allocation.values())
            if not np.isclose(total, 1.0):
                raise ValueError(f"Traffic allocation must sum to 1.0, got {total}")

class SampleSizeCalculator:
    """Calculate required sample sizes for A/B tests."""

    @staticmethod
    def proportions(p1: float, p2: float, alpha: float = 0.05,
                   power: float = 0.80, ratio: float = 1.0) -> Dict:
        """
        Sample size for comparing two proportions.

        Parameters
        ----------
        p1 : float
            Control proportion (baseline)
        p2 : float
            Treatment proportion (alternative)
        alpha : float
            Significance level (default 0.05)
        power : float
            Statistical power (default 0.80)
        ratio : float
            Allocation ratio n2/n1 (default 1.0 for equal allocation)

        Returns
        -------
        dict
            Required sample sizes and related information

        Examples
        --------
        >>> calc = SampleSizeCalculator()
        >>> result = calc.proportions(p1=0.10, p2=0.12, power=0.80)
        >>> print(f"Need {result['n1']} per group")
        """
        z_alpha = stats.norm.ppf(1 - alpha/2)
        z_beta = stats.norm.ppf(power)

        # Variance under H1
        var_sum = p1*(1-p1) + p2*(1-p2)/ratio

        # Effect size
        delta = abs(p2 - p1)

        # Sample size for group 1 (control)
        n1 = ((z_alpha + z_beta)**2 * var_sum) / (delta**2)
        n1 = int(np.ceil(n1))

        # Sample size for group 2 (treatment)
        n2 = int(np.ceil(n1 * ratio))

        return {
            'n1': n1,
            'n2': n2,
            'n_total': n1 + n2,
            'n_per_group': n1 if ratio == 1.0 else None,
            'p1': p1,
            'p2': p2,
            'effect_absolute': delta,
            'effect_relative': delta / p1,
            'alpha': alpha,
            'power': power,
            'ratio': ratio
        }

    @staticmethod
    def means(mu1: float, mu2: float, sigma: float,
             alpha: float = 0.05, power: float = 0.80,
             ratio: float = 1.0) -> Dict:
        """
        Sample size for comparing two means.

        Parameters
        ----------
        mu1 : float
            Control mean
        mu2 : float
            Treatment mean
        sigma : float
            Common standard deviation
        alpha : float
            Significance level
        power : float
            Statistical power
        ratio : float
            Allocation ratio n2/n1

        Returns
        -------
        dict
            Required sample sizes
        """
        z_alpha = stats.norm.ppf(1 - alpha/2)
        z_beta = stats.norm.ppf(power)

        delta = abs(mu2 - mu1)

        # Sample size for group 1
        n1 = ((z_alpha + z_beta)**2 * sigma**2 * (1 + 1/ratio)) / (delta**2)
        n1 = int(np.ceil(n1))
        n2 = int(np.ceil(n1 * ratio))

        return {
            'n1': n1,
            'n2': n2,
            'n_total': n1 + n2,
            'n_per_group': n1 if ratio == 1.0 else None,
            'mu1': mu1,
            'mu2': mu2,
            'sigma': sigma,
            'effect_absolute': delta,
            'effect_standardized': delta / sigma,
            'alpha': alpha,
            'power': power,
            'ratio': ratio
        }


class ABTestAnalyzer:
    """Comprehensive A/B test statistical analysis."""

    def __init__(self, alpha: float = 0.05):
        """
        Initialize analyzer.

        Parameters
        ----------
        alpha : float
            Significance level for hypothesis tests
        """
        self.alpha = alpha

    def analyze_proportions(self,
                           control_conversions: int,
                           control_total: int,
                           treatment_conversions: int,
                           treatment_total: int,
                           method: str = 'z_test',
                           mde: Optional[float] = None) -> ABTestResult:
        """
        Analyze A/B test with binary outcome (conversion).

        Parameters
        ----------
        control_conversions : int
            Number of conversions in control
        control_total : int
            Total users in control
        treatment_conversions : int
            Number of conversions in treatment
        treatment_total : int
            Total users in treatment
        method : str
            Statistical test: 'z_test', 'chi_squared', 'fisher_exact'
        mde : float, optional
            Minimum detectable effect for practical significance

        Returns
        -------
        ABTestResult
            Complete analysis results

        Examples
        --------
        >>> analyzer = ABTestAnalyzer(alpha=0.05)
        >>> result = analyzer.analyze_proportions(
        ...     control_conversions=100,
        ...     control_total=1000,
        ...     treatment_conversions=120,
        ...     treatment_total=1000
        ... )
        >>> print(f"P-value: {result.p_value:.4f}")
        >>> print(f"Lift: {result.relative_lift:.2%}")
        """
        # Calculate proportions
        p1 = control_conversions / control_total
        p2 = treatment_conversions / treatment_total

        # Absolute and relative effects
        abs_diff = p2 - p1
        rel_lift = abs_diff / p1 if p1 > 0 else np.nan

        # Perform statistical test
        if method == 'z_test':
            result = self._z_test_proportions(
                control_conversions, control_total,
                treatment_conversions, treatment_total
            )
        elif method == 'chi_squared':
            result = self._chi_squared_test(
                control_conversions, control_total,
                treatment_conversions, treatment_total
            )
        elif method == 'fisher_exact':
            result = self._fisher_exact_test(
                control_conversions, control_total,
                treatment_conversions, treatment_total
            )
        else:
            raise ValueError(f"Unknown method: {method}")

        # Calculate Cohen's h (effect size for proportions)
        cohens_h = 2 * (np.arcsin(np.sqrt(p2)) - np.arcsin(np.sqrt(p1)))

        # Practical significance
        practically_sig = None
        if mde is not None:
            practically_sig = abs(abs_diff) >= mde

        return ABTestResult(
            metric="conversion_rate",
            control_mean=p1,
            treatment_mean=p2,
            absolute_difference=abs_diff,
            relative_lift=rel_lift,
            standard_error=result['se'],
            confidence_interval=result['ci'],
            p_value=result['p_value'],
            test_statistic=result['test_stat'],
            statistically_significant=result['p_value'] < self.alpha,
            practically_significant=practically_sig,
            sample_size_control=control_total,
            sample_size_treatment=treatment_total,
            cohens_d=cohens_h
        )

    def _z_test_proportions(self, x1, n1, x2, n2) -> Dict:
        """Two-proportion z-test."""
        p1 = x1 / n1
        p2 = x2 / n2

        # Pooled proportion
        p_pool = (x1 + x2) / (n1 + n2)

        # Standard error under null
        se_null = np.sqrt(p_pool * (1 - p_pool) * (1/n1 + 1/n2))

        # Test statistic
        z_stat = (p2 - p1) / se_null
        p_value = 2 * (1 - stats.norm.cdf(abs(z_stat)))

        # Confidence interval (unpooled variance)
        se_unpooled = np.sqrt(p1*(1-p1)/n1 + p2*(1-p2)/n2)
        z_crit = stats.norm.ppf(1 - self.alpha/2)
        ci = (
            (p2 - p1) - z_crit * se_unpooled,
            (p2 - p1) + z_crit * se_unpooled
        )

        return {
            'test_stat': z_stat,
            'p_value': p_value,
            'ci': ci,
            'se': se_unpooled
        }

    def _chi_squared_test(self, x1, n1, x2, n2) -> Dict:
        """Chi-squared test for proportions."""
        table = np.array([[x1, n1 - x1], [x2, n2 - x2]])
        chi2, p_value, dof, expected = stats.chi2_contingency(table)

        # CI from z-test
        p1 = x1 / n1
        p2 = x2 / n2
        se = np.sqrt(p1*(1-p1)/n1 + p2*(1-p2)/n2)
        z_crit = stats.norm.ppf(1 - self.alpha/2)
        ci = ((p2-p1) - z_crit*se, (p2-p1) + z_crit*se)

        return {
            'test_stat': chi2,
            'p_value': p_value,
            'ci': ci,
            'se': se
        }

    def _fisher_exact_test(self, x1, n1, x2, n2) -> Dict:
        """Fisher's exact test."""
        table = np.array([[x1, n1 - x1], [x2, n2 - x2]])
        _, p_value = stats.fisher_exact(table)

        # CI from z-test
        p1 = x1 / n1
        p2 = x2 / n2
        se = np.sqrt(p1*(1-p1)/n1 + p2*(1-p2)/n2)
        z_crit = stats.norm.ppf(1 - self.alpha/2)
        ci = ((p2-p1) - z_crit*se, (p2-p1) + z_crit*se)

        return {
            'test_stat': None,
            'p_value': p_value,
            'ci': ci,
            'se': se
        }

    def analyze_means(self,
                     control_data: np.ndarray,
                     treatment_data: np.ndarray,
                     equal_variance: bool = False,
                     mde: Optional[float] = None) -> ABTestResult:
        """
        Analyze A/B test with continuous outcome.

        Parameters
        ----------
        control_data : array-like
            Control group observations
        treatment_data : array-like
            Treatment group observations
        equal_variance : bool
            Assume equal variances (Student's t) vs unequal (Welch's t)
        mde : float, optional
            Minimum detectable effect

        Returns
        -------
        ABTestResult
            Analysis results
        """
        control = np.asarray(control_data)
        treatment = np.asarray(treatment_data)

        n1 = len(control)
        n2 = len(treatment)
        mean1 = np.mean(control)
        mean2 = np.mean(treatment)
        var1 = np.var(control, ddof=1)
        var2 = np.var(treatment, ddof=1)
        std1 = np.sqrt(var1)
        std2 = np.sqrt(var2)

        # T-test
        t_stat, p_value = stats.ttest_ind(
            treatment, control, equal_var=equal_variance
        )

        # Standard error
        se = np.sqrt(var1/n1 + var2/n2)

        # Degrees of freedom
        if equal_variance:
            df = n1 + n2 - 2
        else:
            # Welch-Satterthwaite
            df = (var1/n1 + var2/n2)**2 / (
                (var1/n1)**2/(n1-1) + (var2/n2)**2/(n2-1)
            )

        # Confidence interval
        t_crit = stats.t.ppf(1 - self.alpha/2, df)
        ci = (
            (mean2 - mean1) - t_crit * se,
            (mean2 - mean1) + t_crit * se
        )

        # Cohen's d
        if equal_variance:
            pooled_std = np.sqrt(((n1-1)*var1 + (n2-1)*var2) / (n1+n2-2))
            cohens_d = (mean2 - mean1) / pooled_std
        else:
            cohens_d = (mean2 - mean1) / np.sqrt((var1 + var2) / 2)

        # Practical significance
        practically_sig = None
        if mde is not None:
            practically_sig = abs(mean2 - mean1) >= mde

        return ABTestResult(
            metric="continuous_metric",
            control_mean=mean1,
            treatment_mean=mean2,
            absolute_difference=mean2 - mean1,
            relative_lift=(mean2 - mean1) / mean1 if mean1 != 0 else np.nan,
            standard_error=se,
            confidence_interval=ci,
            p_value=p_value,
            test_statistic=t_stat,
            statistically_significant=p_value < self.alpha,
            practically_significant=practically_sig,
            sample_size_control=n1,
            sample_size_treatment=n2,
            cohens_d=cohens_d
        )


class DataQualityChecker:
    """Pre-analysis data quality checks."""

    @staticmethod
    def check_sample_ratio_mismatch(data: pd.DataFrame,
                                   variant_col: str = 'variant',
                                   expected_allocation: Optional[Dict] = None,
                                   alpha: float = 0.01) -> Dict:
        """
        Check for Sample Ratio Mismatch (SRM).

        Parameters
        ----------
        data : DataFrame
            Experiment data
        variant_col : str
            Column name for variant assignment
        expected_allocation : dict, optional
            Expected traffic allocation {variant: proportion}
        alpha : float
            Significance level (stricter than test alpha)

        Returns
        -------
        dict
            SRM check results
        """
        observed = data[variant_col].value_counts()
        total = len(data)

        if expected_allocation is None:
            # Assume equal allocation
            n_variants = len(observed)
            expected = {v: total / n_variants for v in observed.index}
        else:
            expected = {v: total * p for v, p in expected_allocation.items()}

        # Chi-square test
        obs_counts = np.array([observed[v] for v in expected.keys()])
        exp_counts = np.array([expected[v] for v in expected.keys()])

        chi2_stat, p_value = stats.chisquare(obs_counts, exp_counts)

        return {
            'passed': p_value >= alpha,
            'p_value': p_value,
            'chi2_statistic': chi2_stat,
            'observed_counts': observed.to_dict(),
            'expected_counts': expected,
            'severity': 'CRITICAL' if p_value < 0.001 else
                       'HIGH' if p_value < 0.01 else 'OK'
        }

    @staticmethod
    def check_novelty_effect(data: pd.DataFrame,
                            metric_col: str,
                            variant_col: str,
                            date_col: str,
                            n_periods: int = 7) -> Dict:
        """
        Check for novelty/learning effects over time.

        Parameters
        ----------
        data : DataFrame
            Experiment data with timestamps
        metric_col : str
            Metric to analyze
        variant_col : str
            Variant column
        date_col : str
            Date/timestamp column
        n_periods : int
            Number of time periods to analyze

        Returns
        -------
        dict
            Temporal stability analysis
        """
        data = data.copy()
        data[date_col] = pd.to_datetime(data[date_col])

        # Split into time periods
        data['period'] = pd.cut(
            data[date_col],
            bins=n_periods,
            labels=range(n_periods)
        )

        # Calculate metrics by period and variant
        temporal_metrics = data.groupby(['period', variant_col])[metric_col].agg([
            'mean', 'std', 'count'
        ]).reset_index()

        # Test for trend in treatment effect over time
        control_data = temporal_metrics[
            temporal_metrics[variant_col] == 'control'
        ].set_index('period')['mean']

        treatment_data = temporal_metrics[
            temporal_metrics[variant_col] == 'treatment'
        ].set_index('period')['mean']

        effects = treatment_data - control_data

        # Spearman correlation with time
        periods = np.arange(len(effects))
        corr, p_value = stats.spearmanr(periods, effects)

        return {
            'temporal_data': temporal_metrics,
            'effects_by_period': effects.to_dict(),
            'trend_correlation': corr,
            'trend_p_value': p_value,
            'stable': abs(corr) < 0.5 or p_value > 0.05
        }

class ResultsReporter:
    """Generate formatted reports from analysis results."""

    @staticmethod
    def format_result(result: ABTestResult,
                     metric_name: str = None,
                     metric_format: str = '.4f') -> str:
        """
        Format AB test result as readable report.

        Parameters
        ----------
        result : ABTestResult
            Analysis results
        metric_name : str, optional
            Display name for metric
        metric_format : str
            Number format string

        Returns
        -------
        str
            Formatted report
        """
        metric_name = metric_name or result.metric

        report = f"""
{'='*70}
A/B TEST RESULTS: {metric_name.upper()}
{'='*70}

SAMPLE SIZES
------------
Control:   {result.sample_size_control:,}
Treatment: {result.sample_size_treatment:,}
Total:     {result.sample_size_control + result.sample_size_treatment:,}

METRIC VALUES
-------------
Control Mean:   {result.control_mean:{metric_format}}
Treatment Mean: {result.treatment_mean:{metric_format}}

EFFECT SIZES
------------
Absolute Difference: {result.absolute_difference:{metric_format}}
Relative Lift:       {result.relative_lift:+.2%}
"""

        if result.cohens_d is not None:
            report += f"Cohen's d:           {result.cohens_d:.4f}\n"

        report += f"""
STATISTICAL INFERENCE
--------------------
Standard Error: {result.standard_error:{metric_format}}
95% CI:         [{result.confidence_interval[0]:{metric_format}}, {result.confidence_interval[1]:{metric_format}}]
Test Statistic: {result.test_statistic:.4f}
P-value:        {result.p_value:.6f}

DECISION
--------
Statistically Significant: {'YES' if result.statistically_significant else 'NO'}
"""

        if result.practically_significant is not None:
            report += f"Practically Significant:   {'YES' if result.practically_significant else 'NO'}\n"

        report += f"\n{'='*70}\n"

        return report

    @staticmethod
    def create_summary_table(results: List[ABTestResult],
                           metric_names: List[str]) -> pd.DataFrame:
        """
        Create summary table from multiple results.

        Parameters
        ----------
        results : list of ABTestResult
            Multiple test results
        metric_names : list of str
            Names for each metric

        Returns
        -------
        DataFrame
            Summary table
        """
        rows = []
        for result, name in zip(results, metric_names):
            rows.append({
                'Metric': name,
                'Control': f"{result.control_mean:.4f}",
                'Treatment': f"{result.treatment_mean:.4f}",
                'Difference': f"{result.absolute_difference:.4f}",
                'Lift': f"{result.relative_lift:+.2%}",
                'CI_Lower': f"{result.confidence_interval[0]:.4f}",
                'CI_Upper': f"{result.confidence_interval[1]:.4f}",
                'P-value': f"{result.p_value:.4f}",
                'Significant': 'Yes' if result.statistically_significant else 'No'
            })

        return pd.DataFrame(rows)

#' Calculate sample size for proportion test
#'
#' @param p1 Control proportion
#' @param p2 Treatment proportion
#' @param alpha Significance level (default 0.05)
#' @param power Statistical power (default 0.80)
#' @param ratio Allocation ratio n2/n1 (default 1.0)
#'
#' @return List with sample size requirements
#'
#' @examples
#' calculate_sample_size_proportions(p1 = 0.10, p2 = 0.12)
#'
#' @export
calculate_sample_size_proportions <- function(p1, p2, alpha = 0.05,
                                              power = 0.80, ratio = 1.0) {

  # Effect size (Cohen's h)
  h <- 2 * (asin(sqrt(p2)) - asin(sqrt(p1)))

  # Sample size using pwr package
  result <- pwr.2p.test(
    h = h,
    sig.level = alpha,
    power = power,
    alternative = "two.sided"
  )

  n_per_group <- ceiling(result$n)

  list(
    n1 = n_per_group,
    n2 = ceiling(n_per_group * ratio),
    n_total = n_per_group * (1 + ratio),
    n_per_group = n_per_group,
    p1 = p1,
    p2 = p2,
    effect_absolute = p2 - p1,
    effect_relative = (p2 - p1) / p1,
    effect_size_h = h,
    alpha = alpha,
    power = power
  )
}


#' Calculate sample size for means test
#'
#' @param mu1 Control mean
#' @param mu2 Treatment mean
#' @param sigma Common standard deviation
#' @param alpha Significance level
#' @param power Statistical power
#' @param ratio Allocation ratio
#'
#' @return List with sample size requirements
#'
#' @export
calculate_sample_size_means <- function(mu1, mu2, sigma,
                                       alpha = 0.05, power = 0.80,
                                       ratio = 1.0) {

  # Effect size (Cohen's d)
  d <- abs(mu2 - mu1) / sigma

  # Sample size
  result <- pwr.t.test(
    d = d,
    sig.level = alpha,
    power = power,
    type = "two.sample",
    alternative = "two.sided"
  )

  n_per_group <- ceiling(result$n)

  list(
    n1 = n_per_group,
    n2 = ceiling(n_per_group * ratio),
    n_total = n_per_group * (1 + ratio),
    n_per_group = n_per_group,
    mu1 = mu1,
    mu2 = mu2,
    sigma = sigma,
    effect_absolute = mu2 - mu1,
    effect_standardized = d,
    alpha = alpha,
    power = power
  )
}

#' Analyze proportions A/B test
#'
#' @param x1 Conversions in control
#' @param n1 Total in control
#' @param x2 Conversions in treatment
#' @param n2 Total in treatment
#' @param alpha Significance level
#' @param method Test method: "prop.test", "chisq", "fisher"
#'
#' @return List with test results
#'
#' @export
analyze_proportions <- function(x1, n1, x2, n2, alpha = 0.05,
                               method = "prop.test") {

  p1 <- x1 / n1
  p2 <- x2 / n2

  # Perform test
  if (method == "prop.test") {
    test <- prop.test(c(x1, x2), c(n1, n2), correct = FALSE)
    p_value <- test$p.value
    ci <- test$conf.int
    test_stat <- test$statistic

  } else if (method == "chisq") {
    table <- matrix(c(x1, n1-x1, x2, n2-x2), nrow = 2, byrow = TRUE)
    test <- chisq.test(table, correct = FALSE)
    p_value <- test$p.value
    test_stat <- test$statistic

    # CI from prop.test
    ci_test <- prop.test(c(x1, x2), c(n1, n2), correct = FALSE)
    ci <- ci_test$conf.int

  } else if (method == "fisher") {
    table <- matrix(c(x1, n1-x1, x2, n2-x2), nrow = 2, byrow = TRUE)
    test <- fisher.test(table)
    p_value <- test$p.value
    test_stat <- NA

    # CI from prop.test
    ci_test <- prop.test(c(x1, x2), c(n1, n2), correct = FALSE)
    ci <- ci_test$conf.int
  }

  # Standard error (unpooled)
  se <- sqrt(p1*(1-p1)/n1 + p2*(1-p2)/n2)

  # Effect sizes
  abs_diff <- p2 - p1
  rel_lift <- abs_diff / p1

  # Cohen's h
  cohens_h <- 2 * (asin(sqrt(p2)) - asin(sqrt(p1)))

  list(
    control_mean = p1,
    treatment_mean = p2,
    absolute_difference = abs_diff,
    relative_lift = rel_lift,
    standard_error = se,
    confidence_interval = ci,
    p_value = p_value,
    test_statistic = test_stat,
    statistically_significant = p_value < alpha,
    sample_size_control = n1,
    sample_size_treatment = n2,
    cohens_h = cohens_h,
    method = method
  )
}


#' Analyze continuous metric A/B test
#'
#' @param control_data Control group values
#' @param treatment_data Treatment group values
#' @param alpha Significance level
#' @param equal_var Assume equal variances (default FALSE)
#'
#' @return List with test results
#'
#' @export
analyze_means <- function(control_data, treatment_data, alpha = 0.05,
                         equal_var = FALSE) {

  # T-test
  test <- t.test(treatment_data, control_data, var.equal = equal_var)

  # Summary statistics
  mean1 <- mean(control_data)
  mean2 <- mean(treatment_data)
  sd1 <- sd(control_data)
  sd2 <- sd(treatment_data)
  n1 <- length(control_data)
  n2 <- length(treatment_data)

  # Effect sizes
  abs_diff <- mean2 - mean1
  rel_lift <- abs_diff / mean1

  # Cohen's d
  if (equal_var) {
    pooled_sd <- sqrt(((n1-1)*sd1^2 + (n2-1)*sd2^2) / (n1+n2-2))
    cohens_d <- abs_diff / pooled_sd
  } else {
    cohens_d <- abs_diff / sqrt((sd1^2 + sd2^2) / 2)
  }

  # Standard error
  se <- sqrt(sd1^2/n1 + sd2^2/n2)

  list(
    control_mean = mean1,
    treatment_mean = mean2,
    absolute_difference = abs_diff,
    relative_lift = rel_lift,
    standard_error = se,
    confidence_interval = test$conf.int,
    p_value = test$p.value,
    test_statistic = test$statistic,
    statistically_significant = test$p.value < alpha,
    sample_size_control = n1,
    sample_size_treatment = n2,
    cohens_d = cohens_d
  )
}

#' Bayesian A/B test for proportions
#'
#' @param x1 Conversions in control
#' @param n1 Total in control
#' @param x2 Conversions in treatment
#' @param n2 Total in treatment
#' @param prior_alpha Prior alpha parameter (default 1)
#' @param prior_beta Prior beta parameter (default 1)
#' @param n_sim Number of Monte Carlo simulations (default 100000)
#'
#' @return List with Bayesian results
#'
#' @examples
#' bayesian_ab_test(x1 = 100, n1 = 1000, x2 = 120, n2 = 1000)
#'
#' @export
bayesian_ab_test <- function(x1, n1, x2, n2,
                             prior_alpha = 1, prior_beta = 1,
                             n_sim = 100000) {

  # Posterior parameters
  post_a1 <- prior_alpha + x1
  post_b1 <- prior_beta + n1 - x1
  post_a2 <- prior_alpha + x2
  post_b2 <- prior_beta + n2 - x2

  # Sample from posteriors
  p1_samples <- rbeta(n_sim, post_a1, post_b1)
  p2_samples <- rbeta(n_sim, post_a2, post_b2)

  # Probability treatment > control
  prob_better <- mean(p2_samples > p1_samples)

  # Expected lift
  lift_samples <- p2_samples / p1_samples - 1
  expected_lift <- mean(lift_samples)
  lift_ci <- quantile(lift_samples, c(0.025, 0.975))

  # Expected loss (opportunity cost)
  loss_control <- mean(pmax(0, p2_samples - p1_samples))
  loss_treatment <- mean(pmax(0, p1_samples - p2_samples))

  # Credible interval for difference
  diff_samples <- p2_samples - p1_samples
  diff_ci <- quantile(diff_samples, c(0.025, 0.975))

  list(
    prob_treatment_better = prob_better,
    expected_lift = expected_lift,
    lift_credible_interval = lift_ci,
    diff_credible_interval = diff_ci,
    loss_choosing_control = loss_control,
    loss_choosing_treatment = loss_treatment,
    recommended_variant = ifelse(loss_control < loss_treatment,
                                 "Control", "Treatment"),
    posterior_samples = data.frame(
      p_control = p1_samples,
      p_treatment = p2_samples
    )
  )
}

#' Check for Sample Ratio Mismatch
#'
#' @param data DataFrame with experiment data
#' @param variant_col Column name for variant
#' @param expected_ratio Expected allocation ratio (named vector)
#' @param alpha Significance level for SRM test (default 0.01)
#'
#' @return List with SRM check results
#'
#' @export
check_sample_ratio_mismatch <- function(data, variant_col = "variant",
                                       expected_ratio = NULL,
                                       alpha = 0.01) {

  observed <- table(data[[variant_col]])
  n_total <- sum(observed)

  if (is.null(expected_ratio)) {
    # Assume equal allocation
    n_variants <- length(observed)
    expected <- rep(n_total / n_variants, n_variants)
    names(expected) <- names(observed)
  } else {
    expected <- expected_ratio * n_total
  }

  # Chi-square test
  test <- chisq.test(observed, p = expected / n_total)

  list(
    passed = test$p.value >= alpha,
    p_value = test$p.value,
    chi_squared = test$statistic,
    observed_counts = observed,
    expected_counts = expected,
    severity = case_when(
      test$p.value < 0.001 ~ "CRITICAL",
      test$p.value < 0.01 ~ "HIGH",
      TRUE ~ "OK"
    )
  )
}


#' Check for temporal stability
#'
#' @param data DataFrame with timestamp column
#' @param metric_col Metric column name
#' @param variant_col Variant column name
#' @param date_col Date/timestamp column name
#' @param n_periods Number of time periods to analyze
#'
#' @return List with temporal analysis
#'
#' @export
check_temporal_stability <- function(data, metric_col, variant_col,
                                    date_col, n_periods = 7) {

  data <- data %>%
    mutate(period = cut(!!sym(date_col), breaks = n_periods, labels = FALSE))

  # Calculate metrics by period
  temporal_stats <- data %>%
    group_by(period, !!sym(variant_col)) %>%
    summarise(
      mean = mean(!!sym(metric_col), na.rm = TRUE),
      sd = sd(!!sym(metric_col), na.rm = TRUE),
      n = n(),
      .groups = 'drop'
    )

  # Calculate treatment effects by period
  effects <- temporal_stats %>%
    select(period, !!sym(variant_col), mean) %>%
    pivot_wider(names_from = !!sym(variant_col), values_from = mean) %>%
    mutate(effect = treatment - control)

  # Test for trend
  cor_test <- cor.test(effects$period, effects$effect, method = "spearman")

  list(
    temporal_data = temporal_stats,
    effects_by_period = effects,
    trend_correlation = cor_test$estimate,
    trend_p_value = cor_test$p.value,
    stable = abs(cor_test$estimate) < 0.5 || cor_test$p.value > 0.05
  )
}

@app.route('/api/v1/experiments', methods=['POST'])
def create_experiment():
    """
    Create a new experiment.

    Request body:
    {
        "name": "Button Color Test",
        "description": "Test green vs blue button",
        "variants": ["control", "treatment"],
        "traffic_allocation": {"control": 0.5, "treatment": 0.5}
    }
    """
    try:
        data = request.get_json()

        # Validate request
        required_fields = ['name', 'description', 'variants', 'traffic_allocation']
        for field in required_fields:
            if field not in data:
                return jsonify({'error': f'Missing required field: {field}'}), 400

        # Validate traffic allocation
        if not abs(sum(data['traffic_allocation'].values()) - 1.0) < 0.001:
            return jsonify({'error': 'Traffic allocation must sum to 1.0'}), 400

        # Generate experiment ID
        exp_id = hashlib.md5(
            f"{data['name']}_{datetime.now().isoformat()}".encode()
        ).hexdigest()[:16]

        # Create experiment
        experiment = Experiment(
            id=exp_id,
            name=data['name'],
            description=data['description'],
            status='draft',
            variants=data['variants'],
            traffic_allocation=data['traffic_allocation'],
            start_date=data.get('start_date'),
            end_date=data.get('end_date')
        )

        # Save to database
        experiments_db[exp_id] = experiment

        return jsonify({
            'success': True,
            'experiment': asdict(experiment)
        }), 201

    except Exception as e:
        return jsonify({'error': str(e)}), 500

@app.route('/api/v1/experiments/<experiment_id>/assign', methods=['POST'])
def assign_variant(experiment_id):
    """
    Assign user to variant.

    Request body:
    {
        "user_id": "user_12345",
        "context": {
            "country": "US",
            "platform": "web"
        }
    }

    Response:
    {
        "variant": "treatment",
        "experiment_id": "exp_123",
        "assignment_timestamp": "2024-01-15T10:30:00"
    }
    """
    try:
        if experiment_id not in experiments_db:
            return jsonify({'error': 'Experiment not found'}), 404

        experiment = experiments_db[experiment_id]

        if experiment.status != 'running':
            return jsonify({'error': 'Experiment is not running'}), 400

        data = request.get_json()

        if 'user_id' not in data:
            return jsonify({'error': 'Missing user_id'}), 400

        user_id = data['user_id']

        # Check if user already assigned
        assignment_key = f"{experiment_id}:{user_id}"
        if assignment_key in assignments_db:
            assignment = assignments_db[assignment_key]
            return jsonify({
                'success': True,
                'variant': assignment['variant'],
                'experiment_id': experiment_id,
                'assignment_timestamp': assignment['timestamp'],
                'is_new_assignment': False
            }), 200

        # Assign variant using deterministic hash
        variant = _assign_variant_deterministic(
            user_id,
            experiment.variants,
            experiment.traffic_allocation,
            experiment.id
        )

        # Store assignment
        assignment = {
            'user_id': user_id,
            'experiment_id': experiment_id,
            'variant': variant,
            'timestamp': datetime.now().isoformat(),
            'context': data.get('context', {})
        }
        assignments_db[assignment_key] = assignment

        return jsonify({
            'success': True,
            'variant': variant,
            'experiment_id': experiment_id,
            'assignment_timestamp': assignment['timestamp'],
            'is_new_assignment': True
        }), 200

    except Exception as e:
        return jsonify({'error': str(e)}), 500


def _assign_variant_deterministic(user_id: str, variants: List[str],
                                  allocation: Dict[str, float],
                                  salt: str) -> str:
    """
    Deterministically assign variant using hash-based randomization.

    Ensures:
    - Same user always gets same variant
    - Traffic split matches desired allocation
    - Cryptographically secure
    """
    # Create hash of user_id + salt
    hash_input = f"{user_id}:{salt}"
    hash_value = int(hashlib.md5(hash_input.encode()).hexdigest(), 16)

    # Normalize to [0, 1)
    normalized = (hash_value % 10000) / 10000.0

    # Assign based on allocation
    cumulative = 0.0
    for variant in variants:
        cumulative += allocation[variant]
        if normalized < cumulative:
            return variant

    # Fallback (should not happen if allocation sums to 1.0)
    return variants[0]

            'Authorization': f'Bearer {api_key}',
            'Content-Type': 'application/json'
        })

    def _request(self, method: str, endpoint: str,
                params: Optional[Dict] = None,
                data: Optional[Dict] = None) -> Dict:
        """
        Make API request.

        Parameters
        ----------
        method : str
            HTTP method (GET, POST, PUT, DELETE)
        endpoint : str
            API endpoint path
        params : dict, optional
            Query parameters
        data : dict, optional
            Request body data

        Returns
        -------
        dict
            Response data

        Raises
        ------
        ABTestAPIError
            If request fails
        """
        url = f"{self.base_url}{endpoint}"

        try:
            response = self.session.request(
                method=method,
                url=url,
                params=params,
                json=data,
                timeout=self.timeout
            )

            response.raise_for_status()
            return response.json()

        except requests.exceptions.HTTPError as e:
            error_msg = f"API request failed: {e}"
            if e.response is not None:
                try:
                    error_data = e.response.json()
                    error_msg = f"API error: {error_data.get('error', str(e))}"
                except:
                    pass
            raise ABTestAPIError(error_msg) from e

        except requests.exceptions.RequestException as e:
            raise ABTestAPIError(f"Request failed: {e}") from e
