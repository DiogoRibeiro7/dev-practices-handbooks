\chapter{Appendix: checklists}

\section{API design checklist}
\begin{itemize}[nosep]
  \item Resource names are nouns.
  \item Errors use a consistent shape.
  \item Pagination is defined.
\end{itemize}

\section{Data modeling checklist}
\begin{itemize}[nosep]
  \item Constraints enforce integrity.
  \item Indexes match query patterns.
  \item Migrations are reversible.
\end{itemize}

\section{Operations checklist}
\begin{itemize}[nosep]
  \item Alerts are actionable.
  \item Runbooks exist for top incidents.
  \item Deployments are automated.
\end{itemize}

\section{Security checklist}
\begin{itemize}[nosep]
  \item Input is validated.
  \item Secrets are rotated.
  \item Dependencies are patched.
\end{itemize}
